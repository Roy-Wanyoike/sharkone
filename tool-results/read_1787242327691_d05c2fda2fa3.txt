   256→// ===================== STATUS STEPPER =====================
   257→
   258→function StatusStepper({ current }: { current: DeliveryStep }) {
   259→  const currentIndex = allSteps.indexOf(current);
   260→
   261→  return (
   262→    <div className="flex items-center gap-0 w-full mt-4">
   263→      {allSteps.map((step, i) => {
   264→        const isCompleted = i < currentIndex;
   265→        const isCurrent = i === currentIndex;
   266→        const isLast = i === allSteps.length - 1;
   267→
   268→        return (
   269→          <div key={step} className="flex items-center flex-1 min-w-0">
   270→            <div className="flex flex-col items-center shrink-0">
   271→              <div
   272→                className={`w-7 h-7 rounded-full flex items-center justify-center text-[10px] font-bold transition-colors ${
   273→                  isCompleted
   274→                    ? 'bg-emerald-500 text-white'
   275→                    : isCurrent
   276→                      ? 'bg-[#F59E0B] text-white ring-4 ring-amber-100'
   277→                      : 'bg-gray-200 text-gray-400'
   278→                }`}
   279→              >
   280→                {isCompleted ? <CheckCircle2 className="h-4 w-4" /> : i + 1}
   281→              </div>
   282→              <span className={`text-[9px] mt-1 text-center leading-tight hidden sm:block ${isCurrent ? 'text-amber-700 font-semibold' : isCompleted ? 'text-emerald-600' : 'text-gray-400'}`}>
   283→                {stepLabels[step]}
   284→              </span>
   285→            </div>
   286→            {!isLast && (
   287→              <div className={`flex-1 h-0.5 mx-1 mt-[-12px] sm:mt-0 ${i < currentIndex ? 'bg-emerald-500' : 'bg-gray-200'}`} />
   288→            )}
   289→          </div>
   290→        );
   291→      })}
   292→    </div>
   293→  );
   294→}

   297→
   298→export default function AdminDeliveriesPage() {
   299→  const queryClient = useQueryClient();
   300→  const [sidebarOpen, setSidebarOpen] = useState(false);
   301→
   302→  // Filter state
   303→  const [statusFilter, setStatusFilter] = useState<string>('ALL');
   304→  const [searchQuery, setSearchQuery] = useState('');
   305→  const [currentPage, setCurrentPage] = useState(1);
   306→
   307→  // Dialog state
   308→  const [selectedDeliveryId, setSelectedDeliveryId] = useState<string | null>(null);
   309→  const [reassignDialogId, setReassignDialogId] = useState<string | null>(null);
   310→  const [autoAssignDialogOpen, setAutoAssignDialogOpen] = useState(false);
   311→
   312→  // Admin note & status override state
   313→  const [adminNotes, setAdminNotes] = useState('');
   314→  const [overrideStatus, setOverrideStatus] = useState('');
   315→
   316→  // Reassign state
   317→  const [selectedDriverId, setSelectedDriverId] = useState('');
   318→
   319→  // ===================== QUERIES =====================
   320→
   321→  const statsQuery = useQuery<DeliveryStats>({
   322→    queryKey: ['admin-delivery-stats'],
   323→    queryFn: () => fetch('/api/admin/deliveries/stats').then(r => r.json()),
   324→  });
   325→
   326→  const deliveriesQuery = useQuery<{ deliveries: DeliveryListItem[]; total: number; page: number; totalPages: number }>({
   327→    queryKey: ['admin-deliveries', statusFilter, searchQuery, currentPage],
   328→    queryFn: () => {
   329→      const params = new URLSearchParams({ page: currentPage.toString() });
   330→      if (statusFilter !== 'ALL') params.set('status', statusFilter);
   331→      if (searchQuery) params.set('search', searchQuery);
   332→      return fetch(`/api/admin/deliveries?${params}`).then(r => r.json());
   333→    },
   334→  });
   335→
   336→  const detailQuery = useQuery<{ delivery: DeliveryDetail }>({
   337→    queryKey: ['admin-delivery-detail', selectedDeliveryId],
   338→    queryFn: () => fetch(`/api/admin/deliveries/${selectedDeliveryId}`).then(r => r.json()),
   339→    enabled: !!selectedDeliveryId,
   340→  });
   341→
   342→  const driversQuery = useQuery<{ users: Driver[] }>({
   343→    queryKey: ['delivery-drivers'],
   344→    queryFn: () => fetch('/api/admin/users?role=DELIVERY').then(r => r.json()),
   345→  });
   346→
   347→  const pendingOrdersQuery = useQuery<{ orders: PendingOrder[] }>({
   348→    queryKey: ['pending-delivery-orders'],
   349→    queryFn: () => fetch('/api/admin/deliveries/pending-orders').then(r => r.json()),
   350→    enabled: autoAssignDialogOpen,
   351→  });
   352→
   353→  // ===================== MUTATIONS =====================
   354→
   355→  const assignMutation = useMutation({
   356→    mutationFn: (orderId: string) =>
   357→      fetch('/api/delivery/assign', {
   358→        method: 'POST',
   359→        headers: { 'Content-Type': 'application/json' },
   360→        body: JSON.stringify({ orderId }),
   361→      }).then(r => {
   362→        if (!r.ok) return r.json().then((e: { error: string }) => { throw new Error(e.error || 'Assignment failed'); });
   363→        return r.json();
   364→      }),
   365→    onSuccess: () => {
   366→      toast.success('Delivery assigned successfully!');
   367→      queryClient.invalidateQueries({ queryKey: ['admin-deliveries'] });
   368→      queryClient.invalidateQueries({ queryKey: ['admin-delivery-stats'] });
   369→      queryClient.invalidateQueries({ queryKey: ['pending-delivery-orders'] });
   370→    },
   371→    onError: (error: Error) => toast.error(error.message),
   372→  });
   373→
   374→  const updateStatusMutation = useMutation({
   375→    mutationFn: ({ id, status, notes }: { id: string; status?: string; notes?: string }) =>
   376→      fetch(`/api/admin/deliveries/${id}`, {
   377→        method: 'PUT',
   378→        headers: { 'Content-Type': 'application/json' },
   379→        body: JSON.stringify({ status, notes }),
   380→      }).then(r => {
   381→        if (!r.ok) return r.json().then((e: { error: string }) => { throw new Error(e.error || 'Update failed'); });
   382→        return r.json();
   383→      }),
   384→    onSuccess: () => {
   385→      toast.success('Delivery updated successfully!');
   386→      queryClient.invalidateQueries({ queryKey: ['admin-deliveries'] });
   387→      queryClient.invalidateQueries({ queryKey: ['admin-delivery-stats'] });
   388→      queryClient.invalidateQueries({ queryKey: ['admin-delivery-detail', selectedDeliveryId] });
   389→      setAdminNotes('');
   390→      setOverrideStatus('');
   391→    },
   392→    onError: (error: Error) => toast.error(error.message),
   393→  });
   394→
   395→  const reassignMutation = useMutation({
   396→    mutationFn: ({ deliveryId, deliveryPersonId }: { deliveryId: string; deliveryPersonId: string }) =>
   397→      fetch(`/api/admin/deliveries/${deliveryId}/reassign`, {
   398→        method: 'POST',
   399→        headers: { 'Content-Type': 'application/json' },
   400→        body: JSON.stringify({ deliveryPersonId }),
   401→      }).then(r => {
   402→        if (!r.ok) return r.json().then((e: { error: string }) => { throw new Error(e.error || 'Reassign failed'); });
   403→        return r.json();
   404→      }),
   405→    onSuccess: () => {
   406→      toast.success('Delivery reassigned successfully!');
   407→      setReassignDialogId(null);
   408→      setSelectedDriverId('');
   409→      queryClient.invalidateQueries({ queryKey: ['admin-deliveries'] });
   410→      if (selectedDeliveryId) {
   411→        queryClient.invalidateQueries({ queryKey: ['admin-delivery-detail', selectedDeliveryId] });
   412→      }
   413→    },
   414→    onError: (error: Error) => toast.error(error.message),
   415→  });
   416→
   417→  // ===================== HANDLERS =====================
   418→
   419→  const handleSearch = useCallback((value: string) => {
   420→    setSearchQuery(value);
   421→    setCurrentPage(1);
   422→  }, []);
   423→
   424→  const handleStatusFilter = useCallback((value: string) => {
   425→    setStatusFilter(value);
   426→    setCurrentPage(1);
   427→  }, []);
   428→
   429→  const handleCopyOtp = (otp: string) => {
   430→    navigator.clipboard.writeText(otp).then(() => {
   431→      toast.success('OTP copied to clipboard');
   432→    }).catch(() => {
   433→      toast.info(`OTP: ${otp}`);
   434→    });
   435→  };
   436→
   437→  const handleOpenDetail = (id: string) => {
   438→    setSelectedDeliveryId(id);
   439→    setAdminNotes('');
   440→    setOverrideStatus('');
   441→  };
   442→
   443→  const handleSaveNotes = () => {
   444→    if (!selectedDeliveryId) return;
   445→    updateStatusMutation.mutate({ id: selectedDeliveryId, notes: adminNotes });
   446→  };
   447→
   448→  const handleOverrideStatus = () => {
   449→    if (!selectedDeliveryId || !overrideStatus) return;
   450→    updateStatusMutation.mutate({ id: selectedDeliveryId, status: overrideStatus });
   451→  };
   452→
   453→  const handleReassign = () => {
   454→    if (!reassignDialogId || !selectedDriverId) return;
   455→    reassignMutation.mutate({ deliveryId: reassignDialogId, deliveryPersonId: selectedDriverId });
   456→  };
   457→
   458→  // ===================== COMPUTED =====================
   459→
   460→  const deliveries = deliveriesQuery.data?.deliveries || [];
   461→  const totalPages = deliveriesQuery.data?.totalPages || 1;
   462→  const total = deliveriesQuery.data?.total || 0;
   463→  const deliveryDetail = detailQuery.data?.delivery || null;
   464→  const pendingOrders = pendingOrdersQuery.data?.orders || [];
   465→
   466→  const statCards = [
   467→    { label: 'Active Deliveries', value: statsQuery.data?.active?.toString() ?? '—', icon: Truck, color: 'bg-amber-500/10 text-amber-600' },
   468→    { label: 'Completed Today', value: statsQuery.data?.completedToday?.toString() ?? '—', icon: CheckCircle2, color: 'bg-emerald-500/10 text-emerald-600' },
   469→    { label: 'Pending Assignment', value: statsQuery.data?.pendingAssignment?.toString() ?? '—', icon: Clock, color: 'bg-blue-500/10 text-blue-600' },
   470→    { label: 'Failed', value: statsQuery.data?.failed?.toString() ?? '—', icon: PackageX, color: 'bg-red-500/10 text-red-600' },
   471→  ];
   472→
   473→  // ===================== RENDER =====================
   474→
   475→  return (
   476→    <div className="min-h-screen bg-gray-50 flex">
   477→      <Sidebar activeNav="deliveries" isOpen={sidebarOpen} onClose={() => setSidebarOpen(false)} />
   478→
   479→      <main className="flex-1 min-h-screen flex flex-col">
   480→        {/* Top Bar */}
   481→        <header className="sticky top-0 z-30 bg-white border-b border-gray-200 px-4 md:px-6 py-4">
   482→          <div className="flex items-center justify-between">
   483→            <div className="flex items-center gap-3">
   484→              <Button
   485→                variant="ghost"
   486→                size="icon"
   487→                className="lg:hidden"
   488→                onClick={() => setSidebarOpen(true)}
   489→              >
   490→                <Menu className="h-5 w-5" />
   491→              </Button>
   492→              <div>
   493→                <h1 className="text-xl md:text-2xl font-bold text-slate-900">Delivery Management</h1>
   494→                <p className="text-sm text-slate-500 hidden sm:block">Monitor and manage all deliveries</p>
   495→              </div>
   496→            </div>
   497→            <div className="flex items-center gap-2">
   498→              <Button
   499→                variant="outline"
   500→                size="sm"
   501→                className="hidden sm:flex items-center gap-2"
   502→                onClick={() => {
   503→                  queryClient.invalidateQueries({ queryKey: ['admin-deliveries'] });
   504→                  queryClient.invalidateQueries({ queryKey: ['admin-delivery-stats'] });
   505→                }}
   506→              >
   507→                <RefreshCw className={`h-4 w-4 ${deliveriesQuery.isFetching ? 'animate-spin' : ''}`} />
   508→                Refresh
   509→              </Button>
   510→              <Button
   511→                className="bg-[#F59E0B] hover:bg-amber-600 text-[#0F172A] font-semibold"
   512→                onClick={() => setAutoAssignDialogOpen(true)}
   513→              >
   514→                <Zap className="h-4 w-4 mr-2" />
   515→                <span className="hidden sm:inline">Auto-Assign Pending</span>
   516→                <span className="sm:hidden">Assign</span>
   517→              </Button>
   518→            </div>
   519→          </div>
   520→        </header>
   521→
   522→        <div className="p-4 md:p-6 space-y-6 flex-1">
   523→          {/* Stats Row */}
   524→          <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   525→            {statCards.map((stat, i) => {
   526→              const Icon = stat.icon;
   527→              return (
   528→                <motion.div
   529→                  key={stat.label}
   530→                  initial={{ opacity: 0, y: 20 }}
   531→                  animate={{ opacity: 1, y: 0 }}
   532→                  transition={{ delay: i * 0.05 }}
   533→                  className="bg-white rounded-xl border border-gray-200 p-4 md:p-6"
   534→                >
   535→                  <div className="flex items-center justify-between">
   536→                    <div className={`p-2.5 rounded-lg ${stat.color}`}>
   537→                      <Icon className="h-5 w-5" />
   538→                    </div>
   539→                  </div>
   540→                  <div className="mt-3">
   541→                    <p className="text-2xl font-bold text-slate-900">{stat.value}</p>
   542→                    <p className="text-sm text-slate-500 mt-0.5">{stat.label}</p>
   543→                  </div>
   544→                </motion.div>
   545→              );
   546→            })}
   547→          </div>
   548→
   549→          {/* Filter Bar */}
   550→          <div className="bg-white rounded-xl border border-gray-200 p-4">
   551→            <div className="flex flex-col sm:flex-row gap-3">
   552→              <div className="relative flex-1">
   553→                <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-gray-400" />
   554→                <Input
   555→                  placeholder="Search by order number..."
   556→                  value={searchQuery}
   557→                  onChange={(e) => handleSearch(e.target.value)}
   558→                  className="pl-9"
