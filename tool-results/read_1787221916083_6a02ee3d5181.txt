   379→// ===================== ORDER STATUS PIE CHART =====================
   380→function OrderStatusChart({ data }: { data: OrderStatusBreakdown[] }) {
   381→  const formattedData = data.map((d) => ({
   382→    name: d.status.replace(/_/g, ' '),
   383→    value: d.count,
   384→    status: d.status,
   385→  }));
   386→
   387→  const renderCustomLabel = ({ cx, cy, midAngle, innerRadius, outerRadius, percent }: any) => {
   388→    if (percent < 0.05) return null;
   389→    const RADIAN = Math.PI / 180;
   390→    const radius = innerRadius + (outerRadius - innerRadius) * 0.5;
   391→    const x = cx + radius * Math.cos(-midAngle * RADIAN);
   392→    const y = cy + radius * Math.sin(-midAngle * RADIAN);
   393→    return (
   394→      <text x={x} y={y} fill="white" textAnchor="middle" dominantBaseline="central" fontSize={12} fontWeight={600}>
   395→        {`${(percent * 100).toFixed(0)}%`}
   396→      </text>
   397→    );
   398→  };
   399→
   400→  return (
   401→    <ResponsiveContainer width="100%" height={300}>
   402→      <PieChart>
   403→        <Pie
   404→          data={formattedData}
   405→          cx="50%"
   406→          cy="50%"
   407→          innerRadius={60}
   408→          outerRadius={100}
   409→          paddingAngle={3}
   410→          dataKey="value"
   411→          label={renderCustomLabel}
   412→          labelLine={false}
   413→        >
   414→          {formattedData.map((_, index) => (
   415→            <Cell key={`cell-${index}`} fill={PIE_COLORS[index % PIE_COLORS.length]} />
   416→          ))}
   417→        </Pie>
   418→        <RechartsTooltip
   419→          contentStyle={{
   420→            backgroundColor: '#fff',
   421→            border: '1px solid #e2e8f0',
   422→            borderRadius: '8px',
   423→            boxShadow: '0 4px 6px -1px rgba(0,0,0,0.1)',
   424→          }}
   425→          formatter={(value: number, name: string) => [value, name]}
   426→        />
   427→        <Legend
   428→          verticalAlign="bottom"
   429→          height={36}
   430→          formatter={(value: string) => (
   431→            <span className="text-xs text-slate-600">{value}</span>
   432→          )}
   433→        />
   434→      </PieChart>
   435→    </ResponsiveContainer>
   436→  );
   437→}
