   141→/* ------------------------------------------------------------------ */
   142→/*  Review Form Component                                              */
   143→/* ------------------------------------------------------------------ */
   144→function ReviewForm({ productId }: { productId: string }) {
   145→  const queryClient = useQueryClient();
   146→  const [rating, setRating] = useState(0);
   147→  const [hoverRating, setHoverRating] = useState(0);
   148→  const [title, setTitle] = useState('');
   149→  const [comment, setComment] = useState('');
   150→  const [userName, setUserName] = useState('');
   151→
   152→  const mutation = useMutation({
   153→    mutationFn: async (data: { rating: number; title?: string; comment: string; userName: string }) => {
   154→      const res = await fetch(`/api/products/${productId}/reviews`, {
   155→        method: 'POST',
   156→        headers: { 'Content-Type': 'application/json' },
   157→        body: JSON.stringify(data),
   158→      });
   159→      if (!res.ok) {
   160→        const body = await res.json().catch(() => ({}));
   161→        throw new Error(body.error || 'Failed to submit review');
   162→      }
   163→      return res.json();
   164→    },
   165→    onSuccess: () => {
   166→      toast.success('Review submitted!');
   167→      queryClient.invalidateQueries({ queryKey: ['reviews', productId] });
   168→      setRating(0);
   169→      setTitle('');
   170→      setComment('');
   171→      setUserName('');
   172→    },
   173→    onError: (err: Error) => {
   174→      toast.error(err.message);
   175→    },
   176→  });
   177→
   178→  const handleSubmit = (e: React.FormEvent) => {
   179→    e.preventDefault();
   180→    if (rating === 0 || !comment.trim() || !userName.trim()) return;
   181→    mutation.mutate({ rating, title: title.trim() || undefined, comment: comment.trim(), userName: userName.trim() });
   182→  };
   183→
   184→  return (
   185→    <Card className="border-gray-100 mb-6">
   186→      <CardContent className="p-5">
   187→        <h3 className="font-bold text-base text-[#0F172A] mb-4">Write a Review</h3>
   188→        <form onSubmit={handleSubmit} className="space-y-4">
   189→          {/* Star selector */}
   190→          <div className="flex items-center gap-1">
   191→            <span className="text-sm font-medium text-gray-700 mr-2">Rating:</span>
   192→            {[1, 2, 3, 4, 5].map((star) => (
   193→              <button
   194→                key={star}
   195→                type="button"
   196→                onClick={() => setRating(star)}
   197→                onMouseEnter={() => setHoverRating(star)}
   198→                onMouseLeave={() => setHoverRating(0)}
   199→                className="p-0.5 transition-transform hover:scale-110"
   200→                aria-label={`Rate ${star} star${star > 1 ? 's' : ''}`}
   201→              >
   202→                <Star
   203→                  className={`h-6 w-6 transition-colors ${
   204→                    star <= (hoverRating || rating)
   205→                      ? 'fill-amber-400 text-amber-400'
   206→                      : 'fill-gray-200 text-gray-200'
   207→                  }`}
   208→                />
   209→              </button>
   210→            ))}
   211→          </div>
   212→          {/* Name */}
   213→          <div>
   214→            <input
   215→              type="text"
   216→              placeholder="Your name"
   217→              value={userName}
   218→              onChange={(e) => setUserName(e.target.value)}
   219→              required
   220→              className="w-full border border-gray-300 rounded-lg px-3 py-2 text-sm focus:outline-none focus:ring-2 focus:ring-amber-400 focus:border-transparent"
   221→            />
   222→          </div>
   223→          {/* Title */}
   224→          <div>
   225→            <input
   226→              type="text"
   227→              placeholder="Review title (optional)"
   228→              value={title}
   229→              onChange={(e) => setTitle(e.target.value)}
   230→              className="w-full border border-gray-300 rounded-lg px-3 py-2 text-sm focus:outline-none focus:ring-2 focus:ring-amber-400 focus:border-transparent"
   231→            />
   232→          </div>
   233→          {/* Comment */}
   234→          <div>
   235→            <textarea
   236→              placeholder="Share your experience with this product..."
   237→              value={comment}
   238→              onChange={(e) => setComment(e.target.value)}
   239→              required
   240→              rows={3}
   241→              className="w-full border border-gray-300 rounded-lg px-3 py-2 text-sm focus:outline-none focus:ring-2 focus:ring-amber-400 focus:border-transparent resize-none"
   242→            />
   243→          </div>
   244→          <Button
   245→            type="submit"
   246→            disabled={mutation.isPending || rating === 0 || !comment.trim() || !userName.trim()}
   247→            className="bg-[#F59E0B] hover:bg-amber-600 text-[#0F172A] font-semibold text-sm disabled:opacity-50 disabled:cursor-not-allowed"
   248→          >
   249→            {mutation.isPending ? 'Submitting...' : 'Submit Review'}
   250→          </Button>
   251→        </form>
   252→      </CardContent>
   253→    </Card>
   254→  );
   255→}

   329→/* ------------------------------------------------------------------ */
   330→/*  Product Image Gallery                                              */
   331→/* ------------------------------------------------------------------ */
   332→function ProductImageGallery({ product }: { product: Product }) {
   333→  let imageList: string[] = [product.image];
   334→
   335→  try {
   336→    if (product.images) {
   337→      const parsed = JSON.parse(product.images);
   338→      if (Array.isArray(parsed) && parsed.length > 0) {
   339→        imageList = parsed;
   340→      }
   341→    }
   342→  } catch {
   343→    // keep default single image
   344→  }
   345→
   346→  const [selectedIdx, setSelectedIdx] = useState(0);
   347→
   348→  return (
   349→    <div className="space-y-4">
   350→      {/* Main image with zoom */}
   351→      <div className="relative aspect-square w-full overflow-hidden rounded-xl border border-gray-100 bg-gray-50 group">
   352→        <motion.img
   353→          key={imageList[selectedIdx]}
   354→          initial={{ opacity: 0 }}
   355→          animate={{ opacity: 1 }}
   356→          transition={{ duration: 0.3 }}
   357→          src={imageList[selectedIdx]}
   358→          alt={product.name}
   359→          className="w-full h-full object-cover transition-transform duration-500 group-hover:scale-110"
   360→        />
   361→        {product.originalPrice && product.originalPrice > product.price && (
   362→          <Badge className="absolute top-3 left-3 bg-red-500 text-white text-xs font-semibold px-2.5 py-1">
   363→            -{Math.round(((product.originalPrice - product.price) / product.originalPrice) * 100)}% OFF
   364→          </Badge>
   365→        )}
   366→      </div>
   367→
   368→      {/* Thumbnails */}
   369→      {imageList.length > 1 && (
   370→        <div className="flex gap-3 overflow-x-auto pb-1">
   371→          {imageList.map((img, idx) => (
   372→            <button
   373→              key={idx}
   374→              onClick={() => setSelectedIdx(idx)}
   375→              className={`shrink-0 w-20 h-20 rounded-lg overflow-hidden border-2 transition-all ${
   376→                idx === selectedIdx
   377→                  ? 'border-amber-500 ring-2 ring-amber-200'
   378→                  : 'border-gray-200 hover:border-gray-400'
   379→              }`}
   380→            >
   381→              <img
   382→                src={img}
   383→                alt={`${product.name} ${idx + 1}`}
   384→                className="w-full h-full object-cover"
   385→              />
   386→            </button>
   387→          ))}
   388→        </div>
   389→      )}
   390→    </div>
   391→  );
   392→}

   396→/* ------------------------------------------------------------------ */
   397→function ReviewSection({ productId }: { productId: string }) {
   398→  const { data, isLoading } = useQuery({
   399→    queryKey: ['reviews', productId],
   400→    queryFn: async () => {
   401→      const res = await fetch(`/api/products/${productId}/reviews?limit=50`);
   402→      if (!res.ok) throw new Error('Failed to load reviews');
   403→      return res.json() as Promise<{ reviews: { id: string; userName: string; rating: number; title?: string | null; comment: string; isVerified: boolean; createdAt: string }[]; total: number; averageRating: number }>;
   404→    },
   405→  });
   406→
   407→  const reviews = data?.reviews ?? [];
   408→  const total = data?.total ?? 0;
   409→  const averageRating = data?.averageRating ?? 0;
   410→
   411→  return (
   412→    <div>
   413→      <ReviewForm productId={productId} />
   414→
   415→      {/* Summary */}
   416→      <div className="flex items-center gap-3 mb-5">
   417→        <div className="flex items-center gap-1.5">
   418→          <Star className="h-5 w-5 fill-amber-400 text-amber-400" />
   419→          <span className="text-lg font-bold text-[#0F172A]">
   420→            {averageRating > 0 ? averageRating.toFixed(1) : '0.0'}
   421→          </span>
   422→        </div>
   423→        <span className="text-sm text-gray-500">
   424→          Based on {total} review{total !== 1 ? 's' : ''}
   425→        </span>
   426→      </div>
   427→
   428→      {/* Review list */}
   429→      {isLoading ? (
   430→        <div className="space-y-4">
   431→          {[1, 2, 3].map((i) => (
   432→            <Card key={i} className="border-gray-100">
   433→              <CardContent className="p-5">
   434→                <div className="flex items-center gap-3">
   435→                  <Skeleton className="h-10 w-10 rounded-full" />
   436→                  <div className="space-y-2">
   437→                    <Skeleton className="h-4 w-32" />
   438→                    <Skeleton className="h-3 w-24" />
   439→                  </div>

   441→                <Skeleton className="h-4 w-full mt-3" />

   444→            </Card>
   445→          ))}
   446→        </div>
   447→      ) : reviews.length === 0 ? (
   448→        <p className="text-sm text-gray-500 text-center py-8">No reviews yet. Be the first to share your experience!</p>
   449→      ) : (
   450→        <div className="space-y-4 max-h-[500px] overflow-y-auto pr-2">
   451→          {reviews.map((review) => (
   452→            <Card key={review.id} className="border-gray-100">
   453→              <CardContent className="p-5">
   454→                <div className="flex items-start justify-between">
   455→                  <div className="flex items-center gap-3">
   456→                    <Avatar className="h-10 w-10 bg-[#0F172A]">
   457→                      <AvatarFallback className="text-white text-sm font-semibold">
   458→                        {review.userName
   459→                          .split(' ')
   460→                          .map((n) => n[0])
   461→                          .join('')
   462→                          .slice(0, 2)
   463→                          .toUpperCase()}
   464→                      </AvatarFallback>
   465→                    </Avatar>
   466→                    <div>
   467→                      <div className="flex items-center gap-2">
   468→                        <p className="font-semibold text-sm text-[#0F172A]">
   469→                          {review.userName}
   470→                        </p>

   473→                            Verified
   474→                          </Badge>

   477→                      <p className="text-xs text-gray-400">
   478→                        {formatDistanceToNow(new Date(review.createdAt), { addSuffix: true })}
   479→                      </p>
   480→                    </div>
   481→                  </div>
   482→                  <StarRating rating={review.rating} size={14} />
   483→                </div>

   485→                  <p className="mt-2 text-sm font-semibold text-[#0F172A]">
   486→                    {review.title}
   487→                  </p>
   488→                )}
   489→                <p className="mt-2 text-sm text-gray-600 leading-relaxed">
   490→                  {review.comment}
   491→                </p>
   492→              </CardContent>
   493→            </Card>
   494→          ))}
   495→        </div>
   496→      )}
   497→    </div>
   498→  );
   499→}
