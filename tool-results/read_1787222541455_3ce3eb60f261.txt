   258→// ===================== STATUS STEPPER =====================
   259→
   260→function StatusStepper({ current }: { current: DeliveryStep }) {
   261→  const currentIndex = allSteps.indexOf(current);
   262→
   263→  return (
   264→    <div className="flex items-center gap-0 w-full mt-4">
   265→      {allSteps.map((step, i) => {
   266→        const isCompleted = i < currentIndex;
   267→        const isCurrent = i === currentIndex;
   268→        const isLast = i === allSteps.length - 1;
   269→
   270→        return (
   271→          <div key={step} className="flex items-center flex-1 min-w-0">
   272→            <div className="flex flex-col items-center shrink-0">
   273→              <div
   274→                className={`w-7 h-7 rounded-full flex items-center justify-center text-[10px] font-bold transition-colors ${
   275→                  isCompleted
   276→                    ? 'bg-emerald-500 text-white'
   277→                    : isCurrent
   278→                      ? 'bg-[#F59E0B] text-white ring-4 ring-amber-100'
   279→                      : 'bg-gray-200 text-gray-400'
   280→                }`}
   281→              >
   282→                {isCompleted ? <CheckCircle2 className="h-4 w-4" /> : i + 1}
   283→              </div>
   284→              <span className={`text-[9px] mt-1 text-center leading-tight hidden sm:block ${isCurrent ? 'text-amber-700 font-semibold' : isCompleted ? 'text-emerald-600' : 'text-gray-400'}`}>
   285→                {stepLabels[step]}
   286→              </span>
   287→            </div>
   288→            {!isLast && (
   289→              <div className={`flex-1 h-0.5 mx-1 mt-[-12px] sm:mt-0 ${i < currentIndex ? 'bg-emerald-500' : 'bg-gray-200'}`} />
   290→            )}
   291→          </div>
   292→        );
   293→      })}
   294→    </div>
   295→  );
   296→}

   300→export default function AdminDeliveriesPage() {
   301→  const queryClient = useQueryClient();
   302→  const [sidebarOpen, setSidebarOpen] = useState(false);
   303→
   304→  // Filter state
   305→  const [statusFilter, setStatusFilter] = useState<string>('ALL');
   306→  const [searchQuery, setSearchQuery] = useState('');
   307→  const [currentPage, setCurrentPage] = useState(1);
   308→
   309→  // Dialog state
   310→  const [selectedDeliveryId, setSelectedDeliveryId] = useState<string | null>(null);
   311→  const [reassignDialogId, setReassignDialogId] = useState<string | null>(null);
   312→  const [autoAssignDialogOpen, setAutoAssignDialogOpen] = useState(false);
   313→
   314→  // Admin note & status override state
   315→  const [adminNotes, setAdminNotes] = useState('');
   316→  const [overrideStatus, setOverrideStatus] = useState('');
   317→
   318→  // Reassign state
   319→  const [selectedDriverId, setSelectedDriverId] = useState('');
   320→
   321→  // ===================== QUERIES =====================
   322→
   323→  const statsQuery = useQuery<DeliveryStats>({
   324→    queryKey: ['admin-delivery-stats'],
   325→    queryFn: async () => {
   326→      const [active, completedToday, pendingAssignment, failed] = await Promise.all([
   327→        db_count_deliveries('active'),
   328→        db_count_deliveries('completedToday'),
   329→        db_count_deliveries('pendingAssignment'),
   330→        db_count_deliveries('failed'),
   331→      ]);
   332→      return { active, completedToday, pendingAssignment, failed };
   333→    },
   334→  });
   335→
   336→  const deliveriesQuery = useQuery<{ deliveries: DeliveryListItem[]; total: number; page: number; totalPages: number }>({
   337→    queryKey: ['admin-deliveries', statusFilter, searchQuery, currentPage],
   338→    queryFn: () => {
   339→      const params = new URLSearchParams({
   340→        page: currentPage.toString(),
   341→      });
   342→      if (statusFilter !== 'ALL') params.set('status', statusFilter);
   343→      if (searchQuery) params.set('search', searchQuery);
   344→      return fetch(`/api/admin/deliveries?${params}`).then(r => r.json());
   345→    },
   346→  });
   347→
   348→  const detailQuery = useQuery<{ delivery: DeliveryDetail }>({
   349→    queryKey: ['admin-delivery-detail', selectedDeliveryId],
   350→    queryFn: () => fetch(`/api/admin/deliveries/${selectedDeliveryId}`).then(r => r.json()),
   351→    enabled: !!selectedDeliveryId,
   352→  });
   353→
   354→  const driversQuery = useQuery<{ users: Driver[] }>({
   355→    queryKey: ['delivery-drivers'],
   356→    queryFn: () => fetch('/api/admin/users?role=DELIVERY').then(r => r.json()),
   357→  });
   358→
   359→  const pendingOrdersQuery = useQuery<{ orders: PendingOrder[] }>({
   360→    queryKey: ['pending-delivery-orders'],
   361→    queryFn: () => fetch('/api/admin/orders?status=PROCESSING&status=SHIPPED&pageSize=50').then(r => r.json()),
   362→    enabled: autoAssignDialogOpen,
   363→  });
   364→
   365→  // ===================== MUTATIONS =====================
   366→
   367→  const assignMutation = useMutation({
   368→    mutationFn: (orderId: string) =>
   369→      fetch('/api/delivery/assign', {
   370→        method: 'POST',
   371→        headers: { 'Content-Type': 'application/json' },
   372→        body: JSON.stringify({ orderId }),
   373→      }).then(r => {
   374→        if (!r.ok) return r.json().then((e: any) => { throw new Error(e.error || 'Assignment failed'); });
   375→        return r.json();
   376→      }),
   377→    onSuccess: () => {
   378→      toast.success('Delivery assigned successfully!');
   379→      queryClient.invalidateQueries({ queryKey: ['admin-deliveries'] });
   380→      queryClient.invalidateQueries({ queryKey: ['admin-delivery-stats'] });
   381→      queryClient.invalidateQueries({ queryKey: ['pending-delivery-orders'] });
   382→    },
   383→    onError: (error: Error) => toast.error(error.message),
   384→  });
   385→
   386→  const updateStatusMutation = useMutation({
   387→    mutationFn: ({ id, status, notes }: { id: string; status?: string; notes?: string }) =>
   388→      fetch(`/api/admin/deliveries/${id}`, {
   389→        method: 'PUT',
   390→        headers: { 'Content-Type': 'application/json' },
   391→        body: JSON.stringify({ status, notes }),
   392→      }).then(r => {
   393→        if (!r.ok) return r.json().then((e: any) => { throw new Error(e.error || 'Update failed'); });
   394→        return r.json();
   395→      }),
   396→    onSuccess: () => {
   397→      toast.success('Delivery updated successfully!');
   398→      queryClient.invalidateQueries({ queryKey: ['admin-deliveries'] });
   399→      queryClient.invalidateQueries({ queryKey: ['admin-delivery-stats'] });
   400→      queryClient.invalidateQueries({ queryKey: ['admin-delivery-detail', selectedDeliveryId] });
   401→      setAdminNotes('');
   402→      setOverrideStatus('');
   403→    },
   404→    onError: (error: Error) => toast.error(error.message),
   405→  });
   406→
   407→  const reassignMutation = useMutation({
   408→    mutationFn: ({ deliveryId, deliveryPersonId }: { deliveryId: string; deliveryPersonId: string }) =>
   409→      fetch(`/api/admin/deliveries/${deliveryId}/reassign`, {
   410→        method: 'POST',
   411→        headers: { 'Content-Type': 'application/json' },
   412→        body: JSON.stringify({ deliveryPersonId }),
   413→      }).then(r => {
   414→        if (!r.ok) return r.json().then((e: any) => { throw new Error(e.error || 'Reassign failed'); });
   415→        return r.json();
   416→      }),
   417→    onSuccess: () => {
   418→      toast.success('Delivery reassigned successfully!');
   419→      setReassignDialogId(null);
   420→      setSelectedDriverId('');
   421→      queryClient.invalidateQueries({ queryKey: ['admin-deliveries'] });
   422→      queryClient.invalidateQueries({ queryKey: ['admin-delivery-detail', selectedDeliveryId] });
   423→    },
   424→    onError: (error: Error) => toast.error(error.message),
   425→  });
   426→
   427→  // ===================== HANDLERS =====================
   428→
   429→  const handleSearch = useCallback((value: string) => {
   430→    setSearchQuery(value);
   431→    setCurrentPage(1);
   432→  }, []);
   433→
   434→  const handleStatusFilter = useCallback((value: string) => {
   435→    setStatusFilter(value);
   436→    setCurrentPage(1);
   437→  }, []);
   438→
   439→  const handleCopyOtp = (otp: string) => {
   440→    navigator.clipboard.writeText(otp).then(() => {
   441→      toast.success('OTP copied to clipboard');
   442→    }).catch(() => {
   443→      toast.info(`OTP: ${otp}`);
   444→    });
   445→  };
   446→
   447→  const handleOpenDetail = (id: string) => {
   448→    setSelectedDeliveryId(id);
   449→    setAdminNotes('');
   450→    setOverrideStatus('');
   451→  };
   452→
   453→  const handleSaveNotes = () => {
   454→    if (!selectedDeliveryId) return;
   455→    updateStatusMutation.mutate({ id: selectedDeliveryId, notes: adminNotes });
   456→  };
   457→
   458→  const handleOverrideStatus = () => {
   459→    if (!selectedDeliveryId || !overrideStatus) return;
   460→    updateStatusMutation.mutate({ id: selectedDeliveryId, status: overrideStatus });
   461→  };
   462→
   463→  const handleReassign = () => {
   464→    if (!reassignDialogId || !selectedDriverId) return;
   465→    reassignMutation.mutate({ deliveryId: reassignDialogId, deliveryPersonId: selectedDriverId });
   466→  };
   467→
   468→  // ===================== COMPUTED =====================
   469→
   470→  const deliveries = deliveriesQuery.data?.deliveries || [];
   471→  const totalPages = deliveriesQuery.data?.totalPages || 1;
   472→  const total = deliveriesQuery.data?.total || 0;
   473→  const deliveryDetail = detailQuery.data?.delivery || null;
   474→
   475→  const statCards = [
   476→    { label: 'Active Deliveries', value: statsQuery.data?.active?.toString() ?? '—', icon: Truck, color: 'bg-amber-500/10 text-amber-600' },
   477→    { label: 'Completed Today', value: statsQuery.data?.completedToday?.toString() ?? '—', icon: CheckCircle2, color: 'bg-emerald-500/10 text-emerald-600' },
   478→    { label: 'Pending Assignment', value: statsQuery.data?.pendingAssignment?.toString() ?? '—', icon: Clock, color: 'bg-blue-500/10 text-blue-600' },
   479→    { label: 'Failed', value: statsQuery.data?.failed?.toString() ?? '—', icon: PackageX, color: 'bg-red-500/10 text-red-600' },
   480→  ];
   481→
   482→  const pendingOrders = (pendingOrdersQuery.data?.orders || []).filter(
   483→    (o: PendingOrder) => !o.delivery
   484→  );
   485→
   486→  // ===================== RENDER =====================
   487→
   488→  return (
   489→    <div className="min-h-screen bg-gray-50">
   490→      <Sidebar activeNav="deliveries" isOpen={sidebarOpen} onClose={() => setSidebarOpen(false)} />
   491→
   492→      <main className="flex-1 min-h-screen">
   493→        {/* Top Bar */}
   494→        <header className="sticky top-0 z-30 bg-white border-b border-gray-200 px-4 md:px-6 py-4">
   495→          <div className="flex items-center justify-between">
   496→            <div className="flex items-center gap-3">
   497→              <Button
   498→                variant="ghost"
   499→                size="icon"
   500→                className="lg:hidden"
   501→                onClick={() => setSidebarOpen(true)}
   502→              >
   503→                <Menu className="h-5 w-5" />
   504→              </Button>
   505→              <div>
   506→                <h1 className="text-xl md:text-2xl font-bold text-slate-900">Delivery Management</h1>
   507→                <p className="text-sm text-slate-500 hidden sm:block">Monitor and manage all deliveries</p>
   508→              </div>
   509→            </div>
   510→            <div className="flex items-center gap-2">
   511→              <Button
   512→                variant="outline"
   513→                size="sm"
   514→                className="hidden sm:flex items-center gap-2"
   515→                onClick={() => {
   516→                  queryClient.invalidateQueries({ queryKey: ['admin-deliveries'] });
   517→                  queryClient.invalidateQueries({ queryKey: ['admin-delivery-stats'] });
   518→                }}
   519→              >
   520→                <RefreshCw className={`h-4 w-4 ${deliveriesQuery.isFetching ? 'animate-spin' : ''}`} />
   521→                Refresh
   522→              </Button>
   523→              <Button
   524→                className="bg-[#F59E0B] hover:bg-amber-600 text-[#0F172A] font-semibold"
   525→                onClick={() => setAutoAssignDialogOpen(true)}
   526→              >
   527→                <Zap className="h-4 w-4 mr-2" />
   528→                <span className="hidden sm:inline">Auto-Assign Pending</span>
   529→                <span className="sm:hidden">Assign</span>
   530→              </Button>
   531→            </div>
   532→          </div>
   533→        </header>
   534→
   535→        <div className="p-4 md:p-6 space-y-6">
   536→          {/* Stats Row */}
   537→          <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   538→            {statCards.map((stat, i) => {
   539→              const Icon = stat.icon;
   540→              return (
   541→                <motion.div
   542→                  key={stat.label}
   543→                  initial={{ opacity: 0, y: 20 }}
   544→                  animate={{ opacity: 1, y: 0 }}
   545→                  transition={{ delay: i * 0.05 }}
   546→                  className="bg-white rounded-xl border border-gray-200 p-4 md:p-6"
   547→                >
   548→                  <div className="flex items-center justify-between">
   549→                    <div className={`p-2.5 rounded-lg ${stat.color}`}>
   550→                      <Icon className="h-5 w-5" />
   551→                    </div>
   552→                  </div>
   553→                  <div className="mt-3">

   556→                  </div>
   557→                </motion.div>
   558→              );
   559→            })}
   560→          </div>
   561→
