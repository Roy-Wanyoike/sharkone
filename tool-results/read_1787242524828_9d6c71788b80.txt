   179→function RouteMap({ waypoints }: { waypoints: Waypoint[] }) {
   180→  const mounted = useMounted();
   181→  if (!mounted || waypoints.length === 0) return null;
   182→
   183→  const W = 600;
   184→  const H = 320;
   185→  const PAD = 60;
   186→
   187→  const lats = waypoints.map((w) => w.latitude);
   188→  const lngs = waypoints.map((w) => w.longitude);
   189→  const minLat = Math.min(...lats);
   190→  const maxLat = Math.max(...lats);
   191→  const minLng = Math.min(...lngs);
   192→  const maxLng = Math.max(...lngs);
   193→
   194→  const latRange = maxLat - minLat || 0.01;
   195→  const lngRange = maxLng - minLng || 0.01;
   196→
   197→  const toSvgX = (lng: number) => PAD + ((lng - minLng) / lngRange) * (W - 2 * PAD);
   198→  const toSvgY = (lat: number) => PAD + ((maxLat - lat) / latRange) * (H - 2 * PAD);
   199→
   200→  const svgPoints = waypoints.map((w) => ({ x: toSvgX(w.longitude), y: toSvgY(w.latitude) }));
   201→
   202→  // Build a smooth path through all waypoints
   203→  const pathD = svgPoints
   204→    .map((p, i) => `${i === 0 ? 'M' : 'L'} ${p.x} ${p.y}`)
   205→    .join(' ');
   206→
   207→  // Pickup point (first waypoint)
   208→  const pickup = svgPoints[0];
   209→  // Destination is an extrapolated point beyond the last waypoint
   210→  const lastWp = waypoints[waypoints.length - 1];
   211→  const lastPt = svgPoints[svgPoints.length - 1];
   212→  const destX = lastPt.x + Math.min(80, (W - 2 * PAD) * 0.15);
   213→  const destY = lastPt.y - Math.min(30, (H - 2 * PAD) * 0.1);
   214→
   215→  return (
   216→    <div className="bg-[#0F172A] rounded-xl p-4 md:p-6 overflow-hidden">
   217→      <div className="flex items-center justify-between mb-4">
   218→        <h3 className="text-sm font-bold text-white flex items-center gap-2">
   219→          <Navigation className="h-4 w-4 text-amber-400" />
   220→          Live Route Map
   221→        </h3>
   222→        <span className="flex items-center gap-1.5 text-[10px] text-green-400 font-medium bg-green-900/30 px-2.5 py-1 rounded-full">
   223→          <Signal className="h-3 w-3" />
   224→          LIVE
   225→        </span>
   226→      </div>
   227→
   228→      <div className="relative rounded-lg overflow-hidden bg-slate-900 border border-slate-700/50">
   229→        <svg
   230→          viewBox={`0 0 ${W} ${H}`}
   231→          className="w-full h-auto"
   232→          style={{ minHeight: '200px' }}
   233→          aria-label="Delivery route visualization"
   234→        >
   235→          {/* Grid lines for map feel */}
   236→          <defs>
   237→            <pattern id="grid" width="40" height="40" patternUnits="userSpaceOnUse">
   238→              <path d="M 40 0 L 0 0 0 40" fill="none" stroke="#1e293b" strokeWidth="0.5" />
   239→            </pattern>
   240→            <radialGradient id="pulseGlow" cx="50%" cy="50%" r="50%">
   241→              <stop offset="0%" stopColor="#F59E0B" stopOpacity="0.6" />
   242→              <stop offset="100%" stopColor="#F59E0B" stopOpacity="0" />
   243→            </radialGradient>
   244→          </defs>
   245→          <rect width={W} height={H} fill="url(#grid)" />
   246→
   247→          {/* Route path (dotted) */}
   248→          <path
   249→            d={pathD}
   250→            fill="none"
   251→            stroke="#475569"
   252→            strokeWidth="2.5"
   253→            strokeDasharray="8 6"
   254→            strokeLinecap="round"
   255→          />
   256→
   257→          {/* Completed waypoint dots (slate) */}
   258→          {svgPoints.slice(0, -1).map((p, i) => (
   259→            <circle
   260→              key={waypoints[i].id}
   261→              cx={p.x}
   262→              cy={p.y}
   263→              r="4"
   264→              fill="#334155"
   265→              stroke="#64748b"
   266→              strokeWidth="1.5"
   267→            />
   268→          ))}
   269→
   270→          {/* Pickup marker */}
   271→          <g transform={`translate(${pickup.x}, ${pickup.y})`}>
   272→            <circle r="14" fill="#0F172A" stroke="#22c55e" strokeWidth="2" />
   273→            <text textAnchor="middle" dominantBaseline="central" fill="#22c55e" fontSize="12" fontWeight="bold">
   274→              P
   275→            </text>
   276→          </g>
   277→
   278→          {/* Destination marker */}
   279→          <g transform={`translate(${destX}, ${destY})`}>
   280→            <circle r="14" fill="#0F172A" stroke="#ef4444" strokeWidth="2" />
   281→            <text textAnchor="middle" dominantBaseline="central" fill="#ef4444" fontSize="12" fontWeight="bold">
   282→              D
   283→            </text>
   284→            {/* Dashed line from last waypoint to destination */}
   285→          </g>

   288→            y1={lastPt.y}
   289→            x2={destX}
   290→            y2={destY}

   343→function DeliveryTimeline({ status }: { status: string }) {
   344→  const currentIdx = ALL_STATUSES.findIndex((s) => s.key === status);

   346→  return (
   347→    <div className="bg-white rounded-xl p-6 shadow-sm">

   349→      <div className="space-y-0">
   350→        {ALL_STATUSES.map((step, i) => {
   351→          const isLast = i === ALL_STATUSES.length - 1;
   352→          const isCompleted = i < currentIdx || (i === currentIdx && (status === 'DELIVERED' || status === 'FAILED'));
   353→          const isActive = i === currentIdx && status !== 'DELIVERED' && status !== 'FAILED';

   356→          return (
   357→            <motion.div
   358→              key={step.key}
   359→              initial={{ opacity: 0, x: -10 }}

   453→/* ------------------------------------------------------------------ */
   454→/*  Main Track Content (inside Suspense for useSearchParams)           */
   455→/* ------------------------------------------------------------------ */
   456→function TrackContent() {
   457→  const mounted = useMounted();
   458→  const searchParams = useSearchParams();
   459→  const prefillOrder = searchParams.get('order') || '';
   460→
   461→  const [searchInput, setSearchInput] = useState(prefillOrder);
   462→  const [searching, setSearching] = useState(false);
   463→  const [notFound, setNotFound] = useState(false);
   464→
   465→  const [delivery, setDelivery] = useState<TrackingDelivery | null>(null);
   466→  const [trackingInfo, setTrackingInfo] = useState<TrackingInfo | null>(null);
   467→  const [showOtp, setShowOtp] = useState(false);
   468→  const [simulating, setSimulating] = useState(false);
   469→  const simIntervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
   470→
   471→  const fetchTracking = useCallback(async (orderNumber: string) => {
   472→    try {
   473→      const res = await fetch(`/api/delivery/lookup?orderNumber=${encodeURIComponent(orderNumber)}`);
   474→      if (!res.ok) {
   475→        setNotFound(true);
   476→        return null;
   477→      }
   478→      const data = await res.json();
   479→      setDelivery(data.delivery);
   480→      setTrackingInfo(data.trackingInfo);
   481→      setNotFound(false);
   482→      return data.delivery as TrackingDelivery;
   483→    } catch {
   484→      setNotFound(true);
   485→      return null;
   486→    }
   487→  }, []);
   488→
   489→  const handleTrack = async () => {
   490→    if (!searchInput.trim()) return;
   491→    setSearching(true);
   492→    setNotFound(false);
   493→    await fetchTracking(searchInput.trim());
   494→    setSearching(false);
   495→  };
   496→
   497→  const handleRecentTrack = async (orderNumber: string) => {
   498→    setSearchInput(orderNumber);
   499→    setSearching(true);
   500→    setNotFound(false);
   501→    await fetchTracking(orderNumber);
   502→    setSearching(false);
   503→  };
   504→
   505→  // Simulation: add a new waypoint every 5 seconds
   506→  const startSimulation = useCallback(() => {
   507→    if (!delivery) return;
   508→    setSimulating(true);
   509→
   510→    simIntervalRef.current = setInterval(async () => {
   511→      const currentDelivery = await new Promise<TrackingDelivery | null>((resolve) => {
   512→        setDelivery((prev) => {
   513→          resolve(prev);
   514→          return prev;
   515→        });
   516→      });
   517→
   518→      if (!currentDelivery || currentDelivery.waypoints.length === 0) return;
   519→
   520→      const latest = currentDelivery.waypoints[currentDelivery.waypoints.length - 1];
   521→      // Simulate GPS movement: small random offsets trending toward destination
   522→      const latOffset = (Math.random() - 0.4) * 0.002;
   523→      const lngOffset = (Math.random() - 0.35) * 0.003;
   524→      const speed = 25 + Math.random() * 20; // 25-45 km/h
   525→      const heading = latest.heading ? latest.heading + (Math.random() - 0.5) * 15 : 45 + Math.random() * 30;
   526→
   527→      try {
   528→        await fetch(`/api/delivery/${currentDelivery.id}/waypoints`, {
   529→          method: 'POST',
   530→          headers: { 'Content-Type': 'application/json' },
   531→          body: JSON.stringify({
   532→            latitude: latest.latitude + latOffset,
   533→            longitude: latest.longitude + lngOffset,
   534→            speed: Math.round(speed * 10) / 10,
   535→            heading: Math.round(heading * 10) / 10,
   536→          }),
   537→        });
   538→
   539→        // Re-fetch tracking data
   540→        const res = await fetch(`/api/delivery/lookup?orderNumber=${encodeURIComponent(currentDelivery.order.orderNumber)}`);
   541→        if (res.ok) {
   542→          const data = await res.json();
   543→          setDelivery(data.delivery);
   544→          setTrackingInfo(data.trackingInfo);
   545→        }
   546→      } catch {
   547→        // Simulation error - silently continue
   548→      }
   549→    }, 5000);
   550→  }, [delivery]);
   551→
   552→  const stopSimulation = useCallback(() => {
   553→    if (simIntervalRef.current) {
   554→      clearInterval(simIntervalRef.current);
   555→      simIntervalRef.current = null;
   556→    }
   557→    setSimulating(false);
   558→  }, []);
   559→
   560→  // Auto-start simulation when delivery has waypoints and is in transit
   561→  useEffect(() => {

   563→      delivery &&
   564→      delivery.waypoints.length > 0 &&
   565→      (delivery.status === 'IN_TRANSIT' || delivery.status === 'PICKED_UP') &&
   566→      !simulating
   567→    ) {
   568→      // Start GPS simulation in a microtask to avoid synchronous setState in effect
   569→      void Promise.resolve().then(() => startSimulation());
   570→    }
   571→    return () => {
   572→      if (simIntervalRef.current) {
   573→        clearInterval(simIntervalRef.current);
   574→      }
   575→    };
   576→  }, [delivery, simulating, startSimulation]);
   577→
   578→  // Prefill and auto-search if URL has order param
   579→  useEffect(() => {
   580→    if (prefillOrder && mounted) {
   581→      void Promise.resolve().then(() => handleRecentTrack(prefillOrder));

   583→  }, [prefillOrder, mounted]);
   584→
   585→  const hasWaypoints = delivery && delivery.waypoints.length > 0;
   586→
   587→  if (!mounted) {
   588→    return (
   589→      <div className="min-h-screen flex flex-col bg-gray-50">
   590→        <SimpleNavbar />
   591→        <main className="flex-1 flex items-center justify-center">
   592→          <Loader2 className="h-8 w-8 animate-spin text-amber-500" />
   593→        </main>
   594→      </div>
   595→    );
   596→  }
   597→
   598→  return (
   599→    <div className="min-h-screen flex flex-col bg-gray-50">
   600→      <SimpleNavbar />
   601→
   602→      <main className="flex-1">
   603→        {/* Search Section */}
   604→        <section className="bg-[#0F172A] py-16 md:py-20">
   605→          <div className="px-6 md:px-16 lg:px-32">
   606→            <motion.div
   607→              initial={{ opacity: 0, y: 20 }}
   608→              animate={{ opacity: 1, y: 0 }}
   609→              transition={{ duration: 0.5 }}
   610→              className="max-w-2xl mx-auto text-center"
   611→            >
   612→              <Package className="h-10 w-10 text-amber-400 mx-auto mb-4" />
   613→              <h1 className="text-3xl md:text-4xl font-bold text-white">Track Your Order</h1>
   614→              <p className="text-gray-400 mt-2 mb-8">
   615→                Enter your order number to get real-time delivery updates
   616→              </p>
   617→
   618→              <div className="flex gap-3 max-w-lg mx-auto">
   619→                <div className="relative flex-1">
   620→                  <Search className="absolute left-3.5 top-1/2 -translate-y-1/2 h-4 w-4 text-gray-400" />
   621→                  <Input
   622→                    placeholder="SHK-XXXXXX"
   623→                    className="pl-10 h-12 bg-white/10 border-white/20 text-white placeholder:text-gray-500 focus:border-amber-500"
   624→                    value={searchInput}
   625→                    onChange={(e) => setSearchInput(e.target.value)}
   626→                    onKeyDown={(e) => e.key === 'Enter' && handleTrack()}
   627→                  />
   628→                </div>
   629→                <Button
   630→                  onClick={handleTrack}
   631→                  disabled={searching || !searchInput.trim()}
   632→                  className="h-12 px-6 bg-[#F59E0B] hover:bg-amber-600 text-[#0F172A] font-semibold shrink-0"
   633→                >
   634→                  {searching ? (
   635→                    <Loader2 className="h-4 w-4 animate-spin" />
   636→                  ) : (
   637→                    'Track'
   638→                  )}
   639→                </Button>
   640→              </div>
   641→
   642→              {/* Not found message */}
   643→              {notFound && (
   644→                <motion.p
   645→                  initial={{ opacity: 0 }}
   646→                  animate={{ opacity: 1 }}
   647→                  className="text-red-400 text-sm mt-3"
   648→                >
   649→                  No delivery found for this order number. The order may not have been assigned for delivery yet.
   650→                </motion.p>
   651→              )}
   652→
   653→              {/* Recent Orders */}
   654→              {!delivery && !searching && (
   655→                <div className="mt-8">
   656→                  <p className="text-gray-500 text-sm mb-3">Recent tracked orders</p>
   657→                  <div className="flex flex-wrap justify-center gap-2">
   658→                    {[
   659→                      { orderNumber: 'SHK-198452', date: 'Jan 10, 2026' },
   660→                      { orderNumber: 'SHK-175623', date: 'Jan 5, 2026' },
   661→                      { orderNumber: 'SHK-162874', date: 'Dec 28, 2025' },
   662→                    ].map((order) => (
   663→                      <button
   664→                        key={order.orderNumber}
   665→                        onClick={() => handleRecentTrack(order.orderNumber)}

   667→                      >
   668→                        <Clock className="h-3.5 w-3.5 text-gray-500" />
   669→                        <span className="text-gray-300 font-mono text-xs">{order.orderNumber}</span>
   670→                        <span className="text-gray-500 text-xs">{order.date}</span>
   671→                      </button>
   672→                    ))}
   673→                  </div>
   674→                </div>
   675→              )}
   676→            </motion.div>
   677→          </div>
   678→        </section>
   679→
   680→        {/* Tracking Result */}
   681→        <AnimatePresence>
   682→          {delivery && trackingInfo && (
   683→            <motion.section
   684→              initial={{ opacity: 0 }}
   685→              animate={{ opacity: 1 }}
   686→              exit={{ opacity: 0 }}
   687→              className="px-6 md:px-16 lg:px-32 py-10"

   692→                  <div className="flex flex-wrap items-center justify-between gap-3">
   693→                    <div>

   737→                  <div className="lg:col-span-2 space-y-6">
   738→                    {/* Route Map (only when waypoints exist) */}
   739→                    {hasWaypoints && <RouteMap waypoints={delivery.waypoints} />}
   740→
   741→                    {/* Delivery Timeline */}
   742→                    <DeliveryTimeline status={delivery.status} />
   743→
   744→                    {/* Order Items */}
   745→                    {delivery.order.items.length > 0 && (
   746→                      <div className="bg-white rounded-xl p-6 shadow-sm">
   747→                        <h2 className="text-lg font-bold text-[#0F172A] mb-4">Order Items</h2>

   749→                          {delivery.order.items.map((item) => (

   752→                                <div className="w-10 h-10 rounded-lg bg-gray-100 flex items-center justify-center overflow-hidden">

   797→                              .split(' ')

   803→                              {delivery.deliveryPerson.name}
   804→                            </p>
   805→                            {delivery.deliveryPerson.phone && (

   809→                              </div>

   811→                          </div>
   812→                        </div>
   813→                      </div>

   816→                    {/* Shipping Address */}
