   453→   453→function ProductsTab({ onAddProduct }: { onAddProduct: () => void }) {

   455→   455→  const [statusFilter, setStatusFilter] = useState('ALL');

   459→   459→    queryFn: () => fetch(`/api/admin/products?search=${search}&status=${statusFilter}`).then((r) => r.json()),

   468→   468→    onError: () => toast.error('Failed to archive product'),
   469→   469→  });
   470→   470→
   471→   471→  return (
   472→   472→    <motion.div
   473→   473→      initial={{ opacity: 0, x: 20 }}
