   169→function statusBadgeColor(status: string) {
   170→  switch (status) {
   171→    case 'DELIVERED': return 'bg-emerald-100 text-emerald-700';

   177→    case 'OUT_FOR_DELIVERY': return 'bg-violet-100 text-violet-700';

   287→
   288→export function SellerDashboard({ sellerId: propSellerId }: SellerDashboardProps) {
   289→  const mounted = useMounted();
   290→  const queryClient = useQueryClient();
   291→
   292→  // Auto-detect seller if not provided
   293→  const autoSellersQuery = useQuery<{ id: string }[]>({
   294→    queryKey: ['sellers-auto-detect'],
   295→    queryFn: () => fetch('/api/admin/sellers').then(r => r.json()),
   296→    enabled: !propSellerId,
   297→  });
   298→
   299→  const sellerId = propSellerId || autoSellersQuery.data?.[0]?.id || '';
   300→  const isLoading = !propSellerId ? autoSellersQuery.isLoading : false;
   301→  const [addProductOpen, setAddProductOpen] = useState(false);
   302→  const [newProduct, setNewProduct] = useState({ name: '', price: '', description: '', category: '', stock: '' });
   303→
   304→  // Products tab filters
   305→  const [productSearch, setProductSearch] = useState('');
   306→  const [productStatusFilter, setProductStatusFilter] = useState('ALL');
   307→
   308→  /* ---- Queries ---- */
   309→
   310→  const statsQuery = useQuery<SellerStats>({
   311→    queryKey: ['seller-stats', sellerId],
   312→    queryFn: () => fetch(`/api/seller/${sellerId}/stats`).then(r => r.json()),
   313→    enabled: !!sellerId,
   314→  });
   315→
   316→  const productsQuery = useQuery<SellerProduct[]>({
   317→    queryKey: ['seller-products', sellerId],
   318→    queryFn: () => fetch(`/api/seller/${sellerId}/products`).then(r => r.json()),
   319→    enabled: !!sellerId,
   320→  });
   321→
   322→  const ordersQuery = useQuery<SellerOrder[]>({
   323→    queryKey: ['seller-orders', sellerId],
   324→    queryFn: () => fetch(`/api/seller/${sellerId}/orders`).then(r => r.json()),
   325→    enabled: !!sellerId,
   326→  });
   327→
   328→  const returnsQuery = useQuery<ReturnsResponse>({
   329→    queryKey: ['seller-returns', sellerId],
   330→    queryFn: () => fetch(`/api/returns?sellerId=${sellerId}&limit=50`).then(r => r.json()),
   331→    enabled: !!sellerId,
   332→  });
   333→
   334→  const categoriesQuery = useQuery<Category[]>({
   335→    queryKey: ['categories'],
   336→    queryFn: () => fetch('/api/categories').then(r => r.json()),
   337→  });
   338→
   339→  /* ---- Mutations ---- */
   340→
   341→  const addProductMutation = useMutation({
   342→    mutationFn: (data: { name: string; price: string; description: string; category: string; stock: string }) =>
   343→      fetch(`/api/seller/${sellerId}/products`, {
   344→        method: 'POST',
   345→        headers: { 'Content-Type': 'application/json' },
   346→        body: JSON.stringify(data),
   347→      }).then(r => r.json()),
   348→    onSuccess: () => {
   349→      queryClient.invalidateQueries({ queryKey: ['seller-products', sellerId] });
   350→      queryClient.invalidateQueries({ queryKey: ['seller-stats', sellerId] });
   351→      toast.success(`"${newProduct.name}" has been added as a draft`);
   352→      setAddProductOpen(false);
   353→      setNewProduct({ name: '', price: '', description: '', category: '', stock: '' });
   354→    },
   355→    onError: () => {
   356→      toast.error('Failed to create product');
   357→    },
   358→  });
   359→
   360→  const updateReturnMutation = useMutation({
   361→    mutationFn: ({ id, status }: { id: string; status: string }) =>
   362→      fetch(`/api/returns/${id}`, {
   363→        method: 'PUT',
   364→        headers: { 'Content-Type': 'application/json' },
   365→        body: JSON.stringify({ status }),
   366→      }).then(r => {
   367→        if (!r.ok) return r.json().then(d => { throw new Error(d.error || 'Update failed'); });
   368→        return r.json();
   369→      }),
   370→    onSuccess: (_data, variables) => {
   371→      queryClient.invalidateQueries({ queryKey: ['seller-returns', sellerId] });
   372→      toast.success(`Return ${variables.status === 'APPROVED' ? 'approved' : 'rejected'} successfully`);
   373→    },
   374→    onError: (err: Error) => {
   375→      toast.error(err.message);
   376→    },
   377→  });
   378→
   379→  const handleAddProduct = () => {
   380→    if (!newProduct.name || !newProduct.price) {
   381→      toast.error('Please fill in product name and price');
   382→      return;
   383→    }
   384→    addProductMutation.mutate(newProduct);
   385→  };
   386→
   387→  const categories = categoriesQuery.data || [];
   388→  const stats = statsQuery.data;
   389→  const products = productsQuery.data || [];
   390→  const orders = ordersQuery.data || [];
   391→  const returnsData = returnsQuery.data?.returns || [];
   392→
   393→  /* ---- Computed: filtered products ---- */
   394→  const filteredProducts = useMemo(() => {
   395→    let result = products;
   396→    if (productStatusFilter !== 'ALL') {
   397→      result = result.filter(p => p.status === productStatusFilter);
   398→    }
   399→    if (productSearch.trim()) {
   400→      const q = productSearch.toLowerCase();
   401→      result = result.filter(p => p.name.toLowerCase().includes(q));
   402→    }
   403→    return result;
   404→  }, [products, productStatusFilter, productSearch]);
   405→
   406→  /* ---- Computed: order status distribution ---- */
   407→  const orderStatusDistribution = useMemo(() => {
   408→    const map: Record<string, number> = {};
   409→    orders.forEach(o => {
   410→      map[o.status] = (map[o.status] || 0) + 1;
   411→    });
   412→    return Object.entries(map).map(([status, count]) => ({ status, count }));
   413→  }, [orders]);
   414→
   415→  /* ---- Computed: return status distribution ---- */
   416→  const returnStatusDistribution = useMemo(() => {
   417→    const map: Record<string, number> = {};
   418→    returnsData.forEach(r => {
   419→      map[r.status] = (map[r.status] || 0) + 1;
   420→    });
   421→    return Object.entries(map).map(([status, count]) => ({ status, count }));
   422→  }, [returnsData]);
   423→
   424→  /* ---- Computed: analytics from orders ---- */
   425→  const earningsByDate = useMemo(() => {
   426→    const map: Record<string, number> = {};
   427→    orders.forEach(o => {
   428→      const dateKey = o.createdAt.slice(0, 10);
   429→      map[dateKey] = (map[dateKey] || 0) + o.total;
   430→    });
   431→    return Object.entries(map)
   432→      .sort(([a], [b]) => a.localeCompare(b))
   433→      .slice(-7);
   434→  }, [orders]);
   435→
   436→  const maxEarning = useMemo(() => Math.max(...earningsByDate.map(([, v]) => v), 1), [earningsByDate]);
   437→
   438→  const topProducts = useMemo(() => {
   439→    return products
   440→      .slice(0, 5)
   441→      .map(p => ({ name: p.name, revenue: p.price * Math.max(0, 20 - products.indexOf(p)) }));
   442→  }, [products]);
   443→
   444→  const maxProductRevenue = useMemo(() => Math.max(...topProducts.map(p => p.revenue), 1), [topProducts]);
   445→
   446→  const statsCards = stats
   447→    ? [
   448→        { label: 'Total Revenue', value: formatCurrency(stats.totalRevenue), change: '+12.5%', icon: <DollarSign className="h-5 w-5" />, color: 'bg-amber-50 text-amber-700' },
   449→        { label: 'Total Orders', value: String(stats.totalOrders), change: '+8.2%', icon: <ShoppingBag className="h-5 w-5" />, color: 'bg-emerald-50 text-emerald-700' },
   450→        { label: 'Products Listed', value: String(stats.totalProducts), change: `+${stats.totalProducts > 0 ? '1' : '0'}`, icon: <Package className="h-5 w-5" />, color: 'bg-gray-100 text-gray-700' },
   451→        { label: 'Rating', value: stats.rating > 0 ? stats.rating.toFixed(1) : 'N/A', change: stats.rating > 0 ? `+${(stats.rating * 0.02).toFixed(1)}` : '', icon: <Star className="h-5 w-5" />, color: 'bg-amber-50 text-amber-700' },
   452→      ]
   453→    : [];
   454→
   455→  /* ---- Quick stats for overview ---- */
   456→  const quickStats = stats
   457→    ? [
   458→        { label: 'Wallet Balance', value: formatCurrency(stats.balance), icon: <Wallet className="h-4 w-4" />, accent: true },
   459→        { label: 'Pending Clearance', value: formatCurrency(stats.pendingClearance), icon: <Clock className="h-4 w-4" />, accent: false },
   460→        { label: 'Total Earnings', value: formatCurrency(stats.totalEarnings), icon: <TrendingUp className="h-4 w-4" />, accent: false },
   461→        { label: 'Total Withdrawn', value: formatCurrency(stats.totalWithdrawn), icon: <ArrowDownRight className="h-4 w-4" />, accent: false },
   462→      ]
   463→    : [];
   464→
   465→  if (!mounted || isLoading) {
   466→    return (
   467→      <div className="px-4 md:px-16 lg:px-32 py-8">
   468→        <div className="mb-8">
   469→          <Skeleton className="h-8 w-56 mb-2" />
   470→          <Skeleton className="h-4 w-72" />
   471→        </div>
   472→        <div className="grid grid-cols-2 lg:grid-cols-4 gap-4 mb-8">
   473→          {Array.from({ length: 4 }).map((_, i) => <StatCardSkeleton key={i} />)}
   474→        </div>
   475→        <TableSkeleton rows={5} />
   476→      </div>
   477→    );
   478→  }
   479→
   480→  return (
   481→    <div className="px-4 md:px-16 lg:px-32 py-8">
   482→      {/* Header */}
   483→      <div className="mb-8">
   484→        <h1 className="text-2xl md:text-3xl font-bold text-[#0F172A]">Seller Dashboard</h1>
   485→        <p className="text-gray-500 text-sm mt-1">Manage your store, products, and earnings</p>
   486→      </div>
   487→
   488→      <Tabs defaultValue="overview" className="w-full">
   489→        <TabsList className="mb-6 bg-gray-100 p-1 rounded-lg">
   490→          <TabsTrigger
   491→            value="overview"
   492→            className="data-[state=active]:bg-[#0F172A] data-[state=active]:text-white rounded-md px-4 py-2 text-sm font-medium transition-all"
   493→          >
   494→            Overview
   495→          </TabsTrigger>
   496→          <TabsTrigger
   497→            value="products"
   498→            className="data-[state=active]:bg-[#0F172A] data-[state=active]:text-white rounded-md px-4 py-2 text-sm font-medium transition-all"
   499→          >
   500→            Products
   501→          </TabsTrigger>
   502→          <TabsTrigger
   503→            value="orders"
   504→            className="data-[state=active]:bg-[#0F172A] data-[state=active]:text-white rounded-md px-4 py-2 text-sm font-medium transition-all"
   505→          >
   506→            Orders
   507→          </TabsTrigger>
   508→          <TabsTrigger
   509→            value="returns"
   510→            className="data-[state=active]:bg-[#0F172A] data-[state=active]:text-white rounded-md px-4 py-2 text-sm font-medium transition-all"
   511→          >
   512→            Returns
   513→            {returnsData.filter(r => r.status === 'PENDING').length > 0 && (
   514→              <span className="ml-1.5 inline-flex items-center justify-center h-5 min-w-5 px-1 rounded-full bg-[#F59E0B] text-[10px] font-bold text-[#0F172A]">
   515→                {returnsData.filter(r => r.status === 'PENDING').length}
   516→              </span>
   517→            )}
   518→          </TabsTrigger>
   519→          <TabsTrigger
   520→            value="analytics"
   521→            className="data-[state=active]:bg-[#0F172A] data-[state=active]:text-white rounded-md px-4 py-2 text-sm font-medium transition-all"
   522→          >
   523→            Analytics
   524→          </TabsTrigger>
   525→        </TabsList>
   526→
   527→        {/* ================================================================ */}
   528→        {/*  TAB 1: OVERVIEW                                                 */}
   529→        {/* ================================================================ */}
   530→        <TabsContent value="overview">
   531→          {/* Stats Cards */}
   532→          {statsQuery.isLoading ? (
   533→            <div className="grid grid-cols-2 lg:grid-cols-4 gap-4 mb-8">
   534→              {Array.from({ length: 4 }).map((_, i) => <StatCardSkeleton key={i} />)}
   535→            </div>
   536→          ) : statsQuery.error ? (
   537→            <div className="bg-red-50 border border-red-200 rounded-xl p-6 mb-8 flex items-center gap-3">
   538→              <AlertCircle className="h-5 w-5 text-red-500" />
   539→              <p className="text-sm text-red-700">Failed to load dashboard stats. Please try again.</p>
   540→            </div>
   541→          ) : (
   542→            <div className="grid grid-cols-2 lg:grid-cols-4 gap-4 mb-8">
   543→              {statsCards.map((stat, i) => (
   544→                <motion.div
   545→                  key={stat.label}
   546→                  initial={{ opacity: 0, y: 20 }}
   547→                  animate={{ opacity: 1, y: 0 }}
   548→                  transition={{ delay: i * 0.05 }}
   549→                  className="bg-white border border-gray-200 rounded-xl p-5 hover:shadow-md transition-shadow"
   550→                >
   551→                  <div className="flex items-center justify-between mb-3">
