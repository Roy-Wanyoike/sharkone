   155→function StepIndicator({ currentStep }: { currentStep: Step }) {
   156→  return (
   157→    <div className="w-full max-w-2xl mx-auto">
   158→      <div className="flex items-center justify-between">
   159→        {STEP_LABELS.map((label, idx) => {
   160→          const step = (idx + 1) as Step;
   161→          const isCompleted = step < currentStep;
   162→          const isActive = step === currentStep;
   163→          const isUpcoming = step > currentStep;
   164→
   165→          return (
   166→            <div key={label} className="flex items-center flex-1 last:flex-none">
   167→              <div className="flex flex-col items-center gap-1.5">
   168→                <div
   169→                  className={`w-9 h-9 rounded-full flex items-center justify-center text-sm font-semibold transition-all duration-300 ${
   170→                    isCompleted
   171→                      ? 'bg-[#0F172A] text-white'
   172→                      : isActive
   173→                        ? 'bg-[#F59E0B] text-[#0F172A] ring-4 ring-amber-100'
   174→                        : 'bg-gray-100 text-gray-400'
   175→                  }`}
   176→                >
   177→                  {isCompleted ? <Check className="h-4 w-4" /> : step}
   178→                </div>
   179→                <span
   180→                  className={`text-xs font-medium hidden sm:block ${
   181→                    isActive ? 'text-[#0F172A]' : isCompleted ? 'text-[#0F172A]' : 'text-gray-400'
   182→                  }`}
   183→                >
   184→                  {label}
   185→                </span>
   186→              </div>
   187→              {idx < STEP_LABELS.length - 1 && (
   188→                <div
   189→                  className={`flex-1 h-0.5 mx-3 mt-[-1.25rem] sm:mt-[-1.25rem] transition-colors duration-300 ${
   190→                    step < currentStep ? 'bg-[#0F172A]' : 'bg-gray-200'
   191→                  }`}
   192→                />
   193→              )}
   194→            </div>
   195→          );
   196→        })}
   197→      </div>
   198→    </div>
   199→  );

   202→function OrderSummarySidebar({
   203→  items,
   204→  deliveryFee,
   205→  couponDiscount,
   206→  couponCode,
   207→  isFreeShipping,
   208→  onCouponApply,
   209→  onCouponRemove,
   210→}: {
   211→  items: CartItem[];
   212→  deliveryFee: number;
   213→  couponDiscount: number;
   214→  couponCode: string | null;
   215→  isFreeShipping: boolean;
   216→  onCouponApply: (discount: number, code: string, isFreeShipping: boolean) => void;
   217→  onCouponRemove: () => void;
   218→}) {
   219→  const subtotal = items.reduce((sum, i) => sum + i.product.price * i.quantity, 0);
   220→  const platformFee = Math.round(subtotal * 0.02);
   221→  const effectiveDeliveryFee = isFreeShipping ? 0 : deliveryFee;
   222→  const total = Math.max(0, subtotal + effectiveDeliveryFee + platformFee - couponDiscount);
   223→
   224→  return (
   225→    <div className="bg-gray-50 rounded-xl p-6 border border-gray-100 sticky top-24">
   226→      <h3 className="font-semibold text-[#0F172A] text-sm mb-4">Order Summary</h3>
   227→      <div className="space-y-3 max-h-64 overflow-y-auto pr-1">
   228→        {items.map((item) => (
   229→          <div key={item.product.id} className="flex items-center gap-3">
   230→            <div className="relative w-12 h-12 rounded-lg overflow-hidden bg-gray-100 shrink-0">
   231→              <Image
   232→                src={item.product.image}
   233→                alt={item.product.name}
   234→                fill
   235→                className="object-cover"
   236→                sizes="48px"
   237→              />
   238→            </div>
   239→            <div className="flex-1 min-w-0">
   240→              <p className="text-sm font-medium text-[#0F172A] truncate">{item.product.name}</p>
   241→              <p className="text-xs text-gray-500">Qty: {item.quantity}</p>
   242→            </div>
   243→            <p className="text-sm font-semibold text-[#0F172A]">
   244→              KES {(item.product.price * item.quantity).toLocaleString()}
   245→            </p>
   246→          </div>
   247→        ))}
   248→      </div>
   249→      <Separator className="my-4" />
   250→      <div className="space-y-2 text-sm">
   251→        <div className="flex justify-between text-gray-600">
   252→          <span>Subtotal</span>
   253→          <span>KES {subtotal.toLocaleString()}</span>
   254→        </div>
   255→        <div className="flex justify-between text-gray-600">
   256→          <span>Delivery</span>
   257→          <span className={isFreeShipping ? 'line-through text-gray-400' : ''}>
   258→            KES {deliveryFee.toLocaleString()}
   259→          </span>
   260→        </div>
   261→        {isFreeShipping && (
   262→          <div className="flex justify-between text-[#F59E0B] font-medium">
   263→            <span>Free Shipping</span>
   264→            <span>−KES {deliveryFee.toLocaleString()}</span>
   265→          </div>
   266→        )}
   267→        <div className="flex justify-between text-gray-600">
   268→          <span>Platform Fee (2%)</span>
   269→          <span>KES {platformFee.toLocaleString()}</span>
   270→        </div>
   271→        {couponDiscount > 0 && (
   272→          <div className="flex justify-between text-green-600 font-medium">
   273→            <span>Coupon Discount ({couponCode})</span>
   274→            <span>−KES {couponDiscount.toLocaleString()}</span>
   275→          </div>
   276→        )}
   277→        <Separator className="!my-3" />
   278→        <div className="flex justify-between font-bold text-[#0F172A]">
   279→          <span>Total</span>
   280→          <span className="text-base">KES {total.toLocaleString()}</span>
   281→        </div>
   282→      </div>
   283→      <div className="mt-4">
   284→        <CouponInput
   285→          onApply={onCouponApply}
   286→          onRemove={onCouponRemove}
   287→          orderTotal={subtotal}
   288→          appliedCode={couponCode}
   289→        />

   293→}
