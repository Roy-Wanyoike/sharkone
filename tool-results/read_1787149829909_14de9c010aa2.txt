   120→function statusBadgeColor(status: string) {
   121→  switch (status) {
   122→    case 'DELIVERED': return 'bg-emerald-100 text-emerald-700';
   123→    case 'SHIPPED': return 'bg-sky-100 text-sky-700';
   124→    case 'PROCESSING': return 'bg-amber-100 text-amber-700';
   125→    case 'PENDING': return 'bg-gray-100 text-gray-700';
   126→    case 'CANCELLED': return 'bg-red-100 text-red-700';
   127→    case 'CONFIRMED': return 'bg-blue-100 text-blue-700';
   128→    case 'OUT_FOR_DELIVERY': return 'bg-violet-100 text-violet-700';
   129→    case 'REFUNDED': return 'bg-orange-100 text-orange-700';
   130→    default: return 'bg-gray-100 text-gray-700';
   131→  }

   236→    queryKey: ['seller-orders', sellerId],

   253→    mutationFn: (data: { name: string; price: string; description: string; category: string; stock: string }) =>
   254→      fetch(`/api/seller/${sellerId}/products`, {
   255→        method: 'POST',
   256→        headers: { 'Content-Type': 'application/json' },
   257→        body: JSON.stringify(data),
   258→      }).then(r => r.json()),

   272→    mutationFn: (amount: number) =>

   322→      ]

   332→

   334→        <TabsList className="mb-6">
   335→          <TabsTrigger value="overview">Overview</TabsTrigger>
   336→          <TabsTrigger value="products">Products</TabsTrigger>
   337→          <TabsTrigger value="orders">Orders</TabsTrigger>
   338→          <TabsTrigger value="wallet">Wallet</TabsTrigger>
   339→        </TabsList>

   341→        {/* ---- Overview Tab ---- */}
   342→        <TabsContent value="overview">
   343→          {statsQuery.isLoading ? (
   344→            <div className="grid grid-cols-2 lg:grid-cols-4 gap-4 mb-8">

   350→              <p className="text-sm text-red-700">Failed to load dashboard stats. Please try again.</p>

   355→                <motion.div
   356→                  key={stat.label}
   357→                  initial={{ opacity: 0, y: 20 }}
   358→                  animate={{ opacity: 1, y: 0 }}
   359→                  transition={{ delay: i * 0.05 }}
   360→                  className="bg-white border border-gray-200 rounded-xl p-5"
