   271→function NewReturnDialog({
   272→  buyerId,
   273→  open,
   274→  onOpenChange,
   275→}: {
   276→  buyerId: string;
   277→  open: boolean;
   278→  onOpenChange: (open: boolean) => void;
   279→}) {
   280→  const queryClient = useQueryClient();
   281→  const [selectedOrderId, setSelectedOrderId] = useState<string>('');
   282→  const [selectedItemId, setSelectedItemId] = useState<string>('');
   283→  const [reason, setReason] = useState<string>('');
   284→  const [description, setDescription] = useState<string>('');
   285→
   286→  const { data: orders = [], isLoading: ordersLoading } = useQuery<BuyerOrder[]>({
   287→    queryKey: ['buyer-delivered-orders', buyerId],
   288→    queryFn: async () => {
   289→      const res = await fetch(`/api/buyer/${buyerId}/orders?status=DELIVERED`);
   290→      const json = await res.json();
   291→      return json.orders || [];
   292→    },
   293→    enabled: open,
   294→  });
   295→
   296→  const selectedOrder = orders.find((o) => o.id === selectedOrderId);
   297→  const orderItems = selectedOrder?.orderItems || [];
   298→
   299→  const createReturn = useMutation({
   300→    mutationFn: async (data: {
   301→      orderId: string;
   302→      orderItemId: string;
   303→      reason: string;
   304→      description: string;
   305→      buyerId: string;
   306→      sellerId: string;
   307→    }) => {
   308→      const res = await fetch('/api/returns', {
   309→        method: 'POST',
   310→        headers: { 'Content-Type': 'application/json' },
   311→        body: JSON.stringify(data),
   312→      });
   313→      if (!res.ok) {
   314→        const err = await res.json();
   315→        throw new Error(err.error || 'Failed to create return request');
   316→      }
   317→      return res.json();
   318→    },
   319→    onSuccess: () => {
   320→      queryClient.invalidateQueries({ queryKey: ['returns', buyerId] });
   321→      toast.success('Return request submitted successfully!');
   322→      setSelectedOrderId('');
   323→      setSelectedItemId('');
   324→      setReason('');
   325→      setDescription('');
   326→      onOpenChange(false);
   327→    },
   328→    onError: (error) => {
   329→      toast.error(error.message);
   330→    },
   331→  });
   332→
   333→  const handleSubmit = () => {
   334→    if (!selectedOrderId || !selectedItemId || !reason) {
   335→      toast.error('Please fill in all required fields');
   336→      return;
   337→    }
   338→    const item = orderItems.find((i) => i.id === selectedItemId);
   339→    if (!item || !item.sellerUserId) return;
   340→
   341→    createReturn.mutate({
   342→      orderId: selectedOrderId,
   343→      orderItemId: selectedItemId,
   344→      reason,
   345→      description,
   346→      buyerId,
   347→      sellerId: item.sellerUserId,
   348→    });
   349→  };
   350→
   351→  const canSubmit = selectedOrderId && selectedItemId && reason && !createReturn.isPending;
   352→
   353→  return (
   354→    <Dialog open={open} onOpenChange={onOpenChange}>
   355→      <DialogContent className="max-w-md max-h-[90vh] overflow-y-auto">
   356→        <DialogHeader>
   357→          <DialogTitle className="text-[#0F172A] flex items-center gap-2">
   358→            <RotateCcw className="h-5 w-5 text-amber-500" />
   359→            Request a Return
   360→          </DialogTitle>
   361→        </DialogHeader>
   362→
   363→        <div className="space-y-4 mt-2">
   364→          {/* Order Selection */}
   365→          <div className="space-y-2">
   366→            <Label>Select Order</Label>
   367→            {ordersLoading ? (
   368→              <Skeleton className="h-10 w-full" />
   369→            ) : orders.length === 0 ? (
   370→              <p className="text-sm text-muted-foreground py-2">No delivered orders available for return.</p>
   371→            ) : (
   372→              <Select value={selectedOrderId} onValueChange={(val) => {
   373→                setSelectedOrderId(val);
   374→                setSelectedItemId('');
   375→              }}>
   376→                <SelectTrigger className="w-full">
   377→                  <SelectValue placeholder="Choose an order" />
   378→                </SelectTrigger>
   379→                <SelectContent className="max-h-48 overflow-y-auto">
   380→                  {orders.map((order) => (
   381→                    <SelectItem key={order.id} value={order.id}>
   382→                      {order.orderNumber} — {KES.format(order.totalAmount)}
   383→                    </SelectItem>
   384→                  ))}
   385→                </SelectContent>
   386→              </Select>
   387→            )}
   388→          </div>
   389→
   390→          {/* Item Selection */}
   391→          {selectedOrder && (
   392→            <motion.div
   393→              initial={{ opacity: 0, height: 0 }}
   394→              animate={{ opacity: 1, height: 'auto' }}
   395→              className="space-y-2"
   396→            >
   397→              <Label>Select Item</Label>
   398→              <Select value={selectedItemId} onValueChange={setSelectedItemId}>
   399→                <SelectTrigger className="w-full">
   400→                  <SelectValue placeholder="Choose an item" />
   401→                </SelectTrigger>
   402→                <SelectContent className="max-h-48 overflow-y-auto">
   403→                  {orderItems.map((item) => (
   404→                    <SelectItem key={item.id} value={item.id}>
   405→                      <div className="flex items-center gap-2">
   406→                        <img
   407→                          src={item.productImage}
   408→                          alt={item.productName}
   409→                          className="h-6 w-6 rounded object-cover"
   410→                        />
   411→                        <span>{item.productName}</span>
   412→                        <span className="text-muted-foreground ml-auto">{KES.format(item.price * item.quantity)}</span>
   413→                      </div>
   414→                    </SelectItem>
   415→                  ))}
   416→                </SelectContent>
   417→              </Select>
   418→            </motion.div>
   419→          )}
   420→
   421→          {/* Reason */}
   422→          <div className="space-y-2">
   423→            <Label>Reason for Return *</Label>
   424→            <Select value={reason} onValueChange={setReason}>
   425→              <SelectTrigger className="w-full">
   426→                <SelectValue placeholder="Select a reason" />
   427→              </SelectTrigger>
   428→              <SelectContent>
   429→                {RETURN_REASONS.map((r) => (
   430→                  <SelectItem key={r} value={r}>{r}</SelectItem>
   431→                ))}
   432→              </SelectContent>
   433→            </Select>
   434→          </div>
   435→
   436→          {/* Description */}
   437→          <div className="space-y-2">
   438→            <Label htmlFor="return-desc">Description (Optional)</Label>
   439→            <Textarea
   440→              id="return-desc"
   441→              placeholder="Provide more details about the issue..."
   442→              value={description}
   443→              onChange={(e) => setDescription(e.target.value)}
   444→              rows={3}
   445→            />
   446→          </div>
   447→
   448→          {/* Refund Preview */}
   449→          {selectedItemId && (
   450→            <motion.div
   451→              initial={{ opacity: 0, y: 5 }}
   452→              animate={{ opacity: 1, y: 0 }}
   453→              className="rounded-lg bg-amber-50 border border-amber-200 p-3 text-sm"
   454→            >
   455→              <p className="text-muted-foreground">Estimated Refund</p>
   456→              <p className="text-lg font-bold text-amber-700">
   457→                {KES.format(
   458→                  (orderItems.find((i) => i.id === selectedItemId)?.price || 0) *
   459→                  (orderItems.find((i) => i.id === selectedItemId)?.quantity || 1)
   460→                )}
   461→              </p>
   462→            </motion.div>
   463→          )}
   464→
   465→          {/* Submit */}
   466→          <Button
   467→            onClick={handleSubmit}
   468→            disabled={!canSubmit}
   469→            className="w-full bg-[#0F172A] hover:bg-slate-800 text-white"
   470→          >
   471→            {createReturn.isPending ? (
   472→              <Loader2 className="h-4 w-4 mr-2 animate-spin" />
   473→            ) : null}
   474→            Submit Return Request
   475→          </Button>
   476→        </div>
   477→      </DialogContent>
   478→    </Dialog>
   479→  );
   480→}

   568→                key={s}
   569→                variant={statusFilter === s ? 'default' : 'outline'}
   570→                size="sm"
   571→                onClick={() => setStatusFilter(s)}
   572→                className={
   573→                  statusFilter === s
   574→                    ? 'bg-[#0F172A] hover:bg-slate-800 text-white'
   575→                    : 'text-[#0F172A] hover:bg-slate-100'
   576→                }
   577→              >
   578→                {s === 'ALL' ? `All (${total})` : s}
   579→              </Button>
   580→            ))}
   581→          </div>
   582→        </div>
   583→
   584→        {/* Loading State */}
   585→        {isLoading && (
   586→          <div className="space-y-4">
   587→            {Array.from({ length: 3 }).map((_, i) => (
   588→              <Card key={i} className="p-4">
   589→                <div className="flex gap-4">
   590→                  <Skeleton className="h-16 w-16 rounded-lg" />
   591→                  <div className="flex-1 space-y-2">
   592→                    <Skeleton className="h-4 w-3/4" />
   593→                    <Skeleton className="h-3 w-1/2" />
   594→                    <Skeleton className="h-3 w-1/3" />
   595→                  </div>
   596→                </div>
   597→              </Card>
   598→            ))}
   599→          </div>
   600→        )}
   601→
   602→        {/* Empty State */}
   603→        {!isLoading && returns.length === 0 && (
   604→          <motion.div
   605→            initial={{ opacity: 0, y: 20 }}
   606→            animate={{ opacity: 1, y: 0 }}
   607→            className="flex flex-col items-center justify-center py-16 text-center"
   608→          >
   609→            <div className="h-20 w-20 rounded-full bg-muted flex items-center justify-center mb-4">
   610→              <RotateCcw className="h-10 w-10 text-muted-foreground" />
   611→            </div>
   612→            <h3 className="text-lg font-semibold text-[#0F172A] mb-1">No returns found</h3>
   613→            <p className="text-sm text-muted-foreground max-w-sm mb-6">
   614→              {statusFilter !== 'ALL'
   615→                ? `You don't have any ${statusFilter.toLowerCase()} returns.`
   616→                : "You haven't made any return requests yet. If you've received a defective or wrong item, you can request a return."}
   617→            </p>
   618→            {statusFilter === 'ALL' && (
   619→              <Button
   620→                onClick={() => setNewReturnOpen(true)}
   621→                className="bg-amber-500 hover:bg-amber-600 text-white gap-2"
   622→              >
   623→                <Plus className="h-4 w-4" />
   624→                Request a Return
   625→              </Button>
   626→            )}
   627→          </motion.div>
   628→        )}
   629→
   630→        {/* Returns List */}
   631→        {!isLoading && returns.length > 0 && (
   632→          <div className="space-y-3">
   633→            <AnimatePresence mode="popLayout">
   634→              {returns.map((item, index) => (
   635→                <motion.div
   636→                  key={item.id}
   637→                  layout
   638→                  initial={{ opacity: 0, y: 10 }}
   639→                  animate={{ opacity: 1, y: 0 }}
   640→                  exit={{ opacity: 0, x: -20 }}
   641→                  transition={{ duration: 0.2, delay: index * 0.05 }}
   642→                >
   643→                  <Card className="overflow-hidden hover:shadow-md transition-shadow">
   644→                    <CardContent className="p-4 sm:p-6">
   645→                      <div className="flex flex-col sm:flex-row gap-4">
   646→                        {/* Product Image */}
   647→                        <img
   648→                          src={item.productImage}
   649→                          alt={item.productName}
   650→                          className="h-20 w-20 sm:h-24 sm:w-24 rounded-lg object-cover border flex-shrink-0"
   651→                        />
   652→
   653→                        {/* Info */}
   654→                        <div className="flex-1 min-w-0 space-y-2">
   655→                          <div className="flex items-start justify-between gap-2">
   656→                            <div>
   657→                              <p className="text-sm font-mono font-semibold text-amber-600">
   658→                                {item.returnNumber}
   659→                              </p>
   660→                              <h3 className="font-semibold text-[#0F172A] truncate">
   661→                                {item.productName}
   662→                              </h3>
   663→                            </div>
   664→                            <StatusBadge status={item.status} />
   665→                          </div>
   666→
   667→                          <div className="flex flex-wrap gap-x-4 gap-y-1 text-sm text-muted-foreground">
   668→                            <span>Order: {item.orderNumber}</span>
   669→                            <span>Reason: {item.reason}</span>
   670→                            <span>{format(new Date(item.createdAt), 'dd MMM yyyy')}</span>
   671→                          </div>
   672→
   673→                          <div className="flex items-center justify-between pt-1">
   674→                            <div className="flex items-center gap-3">
   675→                              <span className="text-lg font-bold text-[#0F172A]">
   676→                                {KES.format(item.refundAmount)}
   677→                              </span>
   678→                              <RefundStatusBadge status={item.refundStatus} />
   679→                            </div>
   680→                            <Button
   681→                              variant="ghost"
   682→                              size="sm"
   683→                              className="text-[#0F172A] hover:bg-slate-100 gap-1.5"
   684→                              onClick={() => setDetailReturnId(item.id)}
   685→                            >
   686→                              <Eye className="h-3.5 w-3.5" />
   687→                              View Details
   688→                            </Button>
   689→                          </div>
   690→                        </div>
   691→                      </div>
   692→                    </CardContent>
   693→                  </Card>
   694→                </motion.div>
   695→              ))}
   696→            </AnimatePresence>
   697→
   698→            {/* Summary Footer */}
   699→            <div className="text-center text-sm text-muted-foreground pt-4">
   700→              Showing {returns.length} of {total} return{total !== 1 ? 's' : ''}
   701→            </div>
   702→          </div>
   703→        )}
   704→      </main>
   705→
   706→      {/* Detail Dialog */}
   707→      <ReturnDetailDialog
   708→        returnId={detailReturnId}
   709→        open={!!detailReturnId}
   710→        onOpenChange={(open) => {
   711→          if (!open) setDetailReturnId(null);
   712→        }}
   713→      />
   714→    </div>
   715→  );
   716→}
