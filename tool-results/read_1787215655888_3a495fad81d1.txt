   247→   247→/*  Product Image Gallery                                              */
   248→   248→/* ------------------------------------------------------------------ */
   249→   249→function ProductImageGallery({ product }: { product: Product }) {
   250→   250→  let imageList: string[] = [product.image];
   251→   251→
   252→   252→  try {
   253→   253→    if (product.images) {
   254→   254→      const parsed = JSON.parse(product.images);
   255→   255→      if (Array.isArray(parsed) && parsed.length > 0) {
   256→   256→        imageList = parsed;
   257→   257→      }
   258→   258→    }
   259→   259→  } catch {
   260→   260→    // keep default single image
   261→   261→  }
   262→   262→
   263→   263→  const [selectedIdx, setSelectedIdx] = useState(0);
   264→   264→
   265→   265→  return (
   266→   266→    <div className="space-y-4">
   267→   267→      {/* Main image with zoom */}
   268→   268→      <div className="relative aspect-square w-full overflow-hidden rounded-xl border border-gray-100 bg-gray-50 group">
   269→   269→        <motion.img
   270→   270→          key={imageList[selectedIdx]}
   271→   271→          initial={{ opacity: 0 }}
   272→   272→          animate={{ opacity: 1 }}
   273→   273→          transition={{ duration: 0.3 }}
   274→   274→          src={imageList[selectedIdx]}
   275→   275→          alt={product.name}
   276→   276→          className="w-full h-full object-cover transition-transform duration-500 group-hover:scale-110"
   277→   277→        />
   278→   278→        {product.originalPrice && product.originalPrice > product.price && (
   279→   279→          <Badge className="absolute top-3 left-3 bg-red-500 text-white text-xs font-semibold px-2.5 py-1">
   280→   280→            -{Math.round(((product.originalPrice - product.price) / product.originalPrice) * 100)}% OFF
   281→   281→          </Badge>
   282→   282→        )}
   283→   283→      </div>
   284→   284→
   285→   285→      {/* Thumbnails */}
   286→   286→      {imageList.length > 1 && (
   287→   287→        <div className="flex gap-3 overflow-x-auto pb-1">
   288→   288→          {imageList.map((img, idx) => (
   289→   289→            <button
   290→   290→              key={idx}
   291→   291→              onClick={() => setSelectedIdx(idx)}
   292→   292→              className={`shrink-0 w-20 h-20 rounded-lg overflow-hidden border-2 transition-all ${
   293→   293→                idx === selectedIdx
   294→   294→                  ? 'border-amber-500 ring-2 ring-amber-200'
   295→   295→                  : 'border-gray-200 hover:border-gray-400'
   296→   296→              }`}
   297→   297→            >
   298→   298→              <img
   299→   299→                src={img}
   300→   300→                alt={`${product.name} ${idx + 1}`}
   301→   301→                className="w-full h-full object-cover"
   302→   302→              />
   303→   303→            </button>
   304→   304→          ))}
   305→   305→        </div>
   306→   306→      )}
