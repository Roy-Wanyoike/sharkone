   198→// ===================== SIDEBAR =====================
   199→function Sidebar({
   200→  activeTab,
   201→  onTabChange,
   202→  isOpen,
   203→  onClose,
   204→}: {
   205→  activeTab: string;
   206→  onTabChange: (tab: string) => void;
   207→  isOpen: boolean;
   208→  onClose: () => void;
   209→}) {
   210→  return (
   211→    <>
   212→      {/* Mobile overlay */}
   213→      <AnimatePresence>
   214→        {isOpen && (
   215→          <motion.div
   216→            initial={{ opacity: 0 }}
   217→            animate={{ opacity: 1 }}
   218→            exit={{ opacity: 0 }}
   219→            className="fixed inset-0 bg-black/50 z-40 lg:hidden"
   220→            onClick={onClose}
   221→          />
   222→        )}
   223→      </AnimatePresence>
   224→
   225→      {/* Sidebar */}
   226→      <aside
   227→        className={`
   228→          fixed top-0 left-0 z-50 h-full w-64 bg-slate-900 flex flex-col

   230→          lg:translate-x-0 lg:static lg:z-auto
   231→          ${isOpen ? 'translate-x-0' : '-translate-x-full'}
   232→        `}
   233→      >
   234→        <div className="p-6 flex items-center justify-between">
   235→          <AdminLogo />
   236→          <Button
   237→            variant="ghost"
   238→            size="icon"
   239→            className="text-white/70 hover:text-white lg:hidden"
   240→            onClick={onClose}
   241→          >
   242→            <X className="h-5 w-5" />
   243→          </Button>
   244→        </div>
   245→
   246→        <nav className="flex-1 px-3 space-y-1">
   247→          {navItems.map((item) => {
   248→            const Icon = item.icon;
   249→            const isActive = 'href' in item ? false : activeTab === item.key;
   250→            const navClassName = `
   251→                  w-full flex items-center gap-3 px-3 py-2.5 rounded-lg text-sm font-medium transition-colors
   252→                  ${isActive
   253→                    ? 'bg-amber-500 text-white'

   255→                  }
   256→                `;
   257→            if ('href' in item) {
   258→              return (
   259→                <Link key={item.key} href={item.href} className={navClassName}>
   260→                  <Icon className="h-5 w-5" />

   262→                </Link>
   263→              );
   264→            }
   265→            return (
   266→              <button
   267→                key={item.key}
   268→                onClick={() => {
   269→                  onTabChange(item.key);
   270→                  onClose();
   271→                }}
   272→                className={navClassName}
   273→              >
   274→                <Icon className="h-5 w-5" />
   275→                {item.label}
   276→              </button>
   277→            );
   278→          })}
   279→        </nav>
   280→
   281→        <div className="p-4 border-t border-slate-700">
   282→          <Link
   283→            href="/"
   284→            className="flex items-center gap-2 text-slate-400 hover:text-white text-sm transition-colors px-3 py-2"
   285→          >
   286→            <ArrowLeft className="h-4 w-4" />
   287→            Back to Store
   288→          </Link>
   289→        </div>
   290→      </aside>
   291→    </>
   292→  );
   293→}

   462→// ===================== PRODUCTS TAB =====================

   465→  const [statusFilter, setStatusFilter] = useState('ALL');
   466→
   467→  const { data, isLoading, refetch } = useQuery<{ products: AdminProduct[]; total: number }>({
   468→    queryKey: ['admin-products', search, statusFilter],
   469→    queryFn: () => fetch(`/api/admin/products?search=${search}&status=${statusFilter}`).then((r) => r.json()),
   470→  });
   471→
   472→  const archiveMutation = useMutation({
   473→    mutationFn: (id: string) => fetch(`/api/admin/products/${id}`, { method: 'DELETE' }).then((r) => r.json()),
   474→    onSuccess: () => {
   475→      toast.success('Product archived');

   481→  return (
   482→    <motion.div
   483→      initial={{ opacity: 0, x: 20 }}

   532→              <p>No products found</p>
   533→            </div>
   534→          ) : (
   535→            <Table>
   536→              <TableHeader>
   537→                <TableRow>
   538→                  <TableHead>Product</TableHead>
   539→                  <TableHead>Seller</TableHead>
   540→                  <TableHead>Category</TableHead>

   889→function ProductDialog({
   890→  open,
   891→  onOpenChange,
   892→}: {
   893→  open: boolean;
   894→  onOpenChange: (open: boolean) => void;
   895→}) {
   896→  const queryClient = useQueryClient();
   897→  const [form, setForm] = useState({
   898→    name: '', description: '', price: '', originalPrice: '', image: '', images: '',
   899→    categoryId: '', sellerId: '', stock: '100', featured: false, status: 'ACTIVE',
   900→  });
   901→  const [submitting, setSubmitting] = useState(false);
   902→
   903→  const { data: categories } = useQuery<Category[]>({
   904→    queryKey: ['categories'],
   905→    queryFn: () => fetch('/api/categories').then((r) => r.json()),
   906→  });
   907→
   908→  const { data: sellers } = useQuery<AdminSeller[]>({
   909→    queryKey: ['admin-sellers-dialog'],
   910→    queryFn: () => fetch('/api/admin/sellers').then((r) => r.json()),
   911→  });
   912→
   913→  const updateField = useCallback((key: string, value: string | boolean) => {
   914→    setForm((prev) => ({ ...prev, [key]: value }));
   915→  }, []);
   916→
   917→  const handleSubmit = async () => {
   918→    if (!form.name || !form.description || !form.price || !form.image || !form.categoryId || !form.sellerId) {
   919→      toast.error('Please fill all required fields');
   920→      return;
   921→    }
   922→
   923→    setSubmitting(true);
   924→    try {
   925→      const res = await fetch('/api/admin/products', {
   926→        method: 'POST',
   927→        headers: { 'Content-Type': 'application/json' },
   928→        body: JSON.stringify(form),
   929→      });
   930→      if (res.ok) {
   931→        toast.success('Product created successfully');
   932→        queryClient.invalidateQueries({ queryKey: ['admin-products'] });
   933→        onOpenChange(false);
   934→        setForm({
   935→          name: '', description: '', price: '', originalPrice: '', image: '', images: '',
   936→          categoryId: '', sellerId: '', stock: '100', featured: false, status: 'ACTIVE',
   937→        });
   938→      } else {
   939→        const err = await res.json();
   940→        toast.error(err.error || 'Failed to create product');
   941→      }
   942→    } catch {
   943→      toast.error('Failed to create product');
   944→    } finally {
   945→      setSubmitting(false);
   946→    }
