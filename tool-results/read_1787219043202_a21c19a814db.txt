   154→   154→function StepIndicator({ currentStep }: { currentStep: Step }) {
   155→   155→  return (
   156→   156→    <div className="w-full max-w-2xl mx-auto">
   157→   157→      <div className="flex items-center justify-between">
   158→   158→        {STEP_LABELS.map((label, idx) => {
   159→   159→          const step = (idx + 1) as Step;
   160→   160→          const isCompleted = step < currentStep;
   161→   161→          const isActive = step === currentStep;
   162→   162→          const isUpcoming = step > currentStep;
   163→   163→
   164→   164→          return (
   165→   165→            <div key={label} className="flex items-center flex-1 last:flex-none">
   166→   166→              <div className="flex flex-col items-center gap-1.5">
   167→   167→                <div
   168→   168→                  className={`w-9 h-9 rounded-full flex items-center justify-center text-sm font-semibold transition-all duration-300 ${
   169→   169→                    isCompleted
   170→   170→                      ? 'bg-[#0F172A] text-white'
   171→   171→                      : isActive
   172→   172→                        ? 'bg-[#F59E0B] text-[#0F172A] ring-4 ring-amber-100'
   173→   173→                        : 'bg-gray-100 text-gray-400'
   174→   174→                  }`}
   175→   175→                >
   176→   176→                  {isCompleted ? <Check className="h-4 w-4" /> : step}
   177→   177→                </div>
   178→   178→                <span
   179→   179→                  className={`text-xs font-medium hidden sm:block ${
   180→   180→                    isActive ? 'text-[#0F172A]' : isCompleted ? 'text-[#0F172A]' : 'text-gray-400'
   181→   181→                  }`}
   182→   182→                >
   183→   183→                  {label}
   184→   184→                </span>
   185→   185→              </div>
   186→   186→              {idx < STEP_LABELS.length - 1 && (
   187→   187→                <div
   188→   188→                  className={`flex-1 h-0.5 mx-3 mt-[-1.25rem] sm:mt-[-1.25rem] transition-colors duration-300 ${
   189→   189→                    step < currentStep ? 'bg-[#0F172A]' : 'bg-gray-200'
   190→   190→                  }`}
   191→   191→                />
   192→   192→              )}
   193→   193→            </div>
   194→   194→          );
   195→   195→        })}
   196→   196→      </div>
