   142→/* ------------------------------------------------------------------ */
   143→/*  Review Form Component                                              */
   144→/* ------------------------------------------------------------------ */
   145→function ReviewForm({ productId }: { productId: string }) {
   146→  const queryClient = useQueryClient();
   147→  const [rating, setRating] = useState(0);
   148→  const [hoverRating, setHoverRating] = useState(0);
   149→  const [title, setTitle] = useState('');
   150→  const [comment, setComment] = useState('');
   151→  const [userName, setUserName] = useState('');
   152→
   153→  const mutation = useMutation({
   154→    mutationFn: async (data: { rating: number; title?: string; comment: string; userName: string }) => {
   155→      const res = await fetch(`/api/products/${productId}/reviews`, {
   156→        method: 'POST',
   157→        headers: { 'Content-Type': 'application/json' },
   158→        body: JSON.stringify(data),
   159→      });
   160→      if (!res.ok) {
   161→        const body = await res.json().catch(() => ({}));
   162→        throw new Error(body.error || 'Failed to submit review');
   163→      }
   164→      return res.json();
   165→    },
   166→    onSuccess: () => {
   167→      toast.success('Review submitted!');
   168→      queryClient.invalidateQueries({ queryKey: ['reviews', productId] });
   169→      setRating(0);
   170→      setTitle('');
   171→      setComment('');
   172→      setUserName('');
   173→    },
   174→    onError: (err: Error) => {
   175→      toast.error(err.message);
   176→    },
   177→  });
   178→
   179→  const handleSubmit = (e: React.FormEvent) => {
   180→    e.preventDefault();
   181→    if (rating === 0 || !comment.trim() || !userName.trim()) return;
   182→    mutation.mutate({ rating, title: title.trim() || undefined, comment: comment.trim(), userName: userName.trim() });
   183→  };
   184→
   185→  return (
   186→    <Card className="border-gray-100 mb-6">
   187→      <CardContent className="p-5">
   188→        <h3 className="font-bold text-base text-[#0F172A] mb-4">Write a Review</h3>
   189→        <form onSubmit={handleSubmit} className="space-y-4">
   190→          {/* Star selector */}
   191→          <div className="flex items-center gap-1">
   192→            <span className="text-sm font-medium text-gray-700 mr-2">Rating:</span>
   193→            {[1, 2, 3, 4, 5].map((star) => (
   194→              <button
   195→                key={star}
   196→                type="button"
   197→                onClick={() => setRating(star)}
   198→                onMouseEnter={() => setHoverRating(star)}
   199→                onMouseLeave={() => setHoverRating(0)}
   200→                className="p-0.5 transition-transform hover:scale-110"
   201→                aria-label={`Rate ${star} star${star > 1 ? 's' : ''}`}
   202→              >
   203→                <Star
   204→                  className={`h-6 w-6 transition-colors ${
   205→                    star <= (hoverRating || rating)
   206→                      ? 'fill-amber-400 text-amber-400'
   207→                      : 'fill-gray-200 text-gray-200'
   208→                  }`}
   209→                />
   210→              </button>
   211→            ))}
   212→          </div>
   213→          {/* Name */}
   214→          <div>
   215→            <input
   216→              type="text"
   217→              placeholder="Your name"
   218→              value={userName}
   219→              onChange={(e) => setUserName(e.target.value)}
   220→              required
   221→              className="w-full border border-gray-300 rounded-lg px-3 py-2 text-sm focus:outline-none focus:ring-2 focus:ring-amber-400 focus:border-transparent"
   222→            />
   223→          </div>
   224→          {/* Title */}
   225→          <div>
   226→            <input
   227→              type="text"
   228→              placeholder="Review title (optional)"
   229→              value={title}
   230→              onChange={(e) => setTitle(e.target.value)}
   231→              className="w-full border border-gray-300 rounded-lg px-3 py-2 text-sm focus:outline-none focus:ring-2 focus:ring-amber-400 focus:border-transparent"
   232→            />
   233→          </div>
   234→          {/* Comment */}
   235→          <div>
   236→            <textarea
   237→              placeholder="Share your experience with this product..."
   238→              value={comment}
   239→              onChange={(e) => setComment(e.target.value)}
   240→              required
   241→              rows={3}
   242→              className="w-full border border-gray-300 rounded-lg px-3 py-2 text-sm focus:outline-none focus:ring-2 focus:ring-amber-400 focus:border-transparent resize-none"
   243→            />
   244→          </div>
   245→          <Button
   246→            type="submit"
   247→            disabled={mutation.isPending || rating === 0 || !comment.trim() || !userName.trim()}
   248→            className="bg-[#F59E0B] hover:bg-amber-600 text-[#0F172A] font-semibold text-sm disabled:opacity-50 disabled:cursor-not-allowed"
   249→          >
   250→            {mutation.isPending ? 'Submitting...' : 'Submit Review'}
   251→          </Button>
   252→        </form>
   253→      </CardContent>
   254→    </Card>
   255→  );
   256→}

   330→/* ------------------------------------------------------------------ */
   331→/*  Product Image Gallery                                              */
   332→/* ------------------------------------------------------------------ */
   333→function ProductImageGallery({ product }: { product: Product }) {
   334→  let imageList: string[] = [product.image];
   335→
   336→  try {
   337→    if (product.images) {
   338→      const parsed = JSON.parse(product.images);
   339→      if (Array.isArray(parsed) && parsed.length > 0) {
   340→        imageList = parsed;
   341→      }
   342→    }
   343→  } catch {
   344→    // keep default single image
   345→  }
   346→
   347→  const [selectedIdx, setSelectedIdx] = useState(0);
   348→
   349→  return (
   350→    <div className="space-y-4">
   351→      {/* Main image with zoom */}
   352→      <div className="relative aspect-square w-full overflow-hidden rounded-xl border border-gray-100 bg-gray-50 group">
   353→        <motion.img
   354→          key={imageList[selectedIdx]}
   355→          initial={{ opacity: 0 }}
   356→          animate={{ opacity: 1 }}
   357→          transition={{ duration: 0.3 }}
   358→          src={imageList[selectedIdx]}
   359→          alt={product.name}
   360→          className="w-full h-full object-cover transition-transform duration-500 group-hover:scale-110"
   361→        />
   362→        {product.originalPrice && product.originalPrice > product.price && (
   363→          <Badge className="absolute top-3 left-3 bg-red-500 text-white text-xs font-semibold px-2.5 py-1">
   364→            -{Math.round(((product.originalPrice - product.price) / product.originalPrice) * 100)}% OFF
   365→          </Badge>
   366→        )}
   367→      </div>
   368→
   369→      {/* Thumbnails */}
   370→      {imageList.length > 1 && (
   371→        <div className="flex gap-3 overflow-x-auto pb-1">
   372→          {imageList.map((img, idx) => (
   373→            <button
   374→              key={idx}
   375→              onClick={() => setSelectedIdx(idx)}
   376→              className={`shrink-0 w-20 h-20 rounded-lg overflow-hidden border-2 transition-all ${
   377→                idx === selectedIdx
   378→                  ? 'border-amber-500 ring-2 ring-amber-200'
   379→                  : 'border-gray-200 hover:border-gray-400'
   380→              }`}
   381→            >
   382→              <img
   383→                src={img}
   384→                alt={`${product.name} ${idx + 1}`}
   385→                className="w-full h-full object-cover"
   386→              />
   387→            </button>
   388→          ))}
   389→        </div>
   390→      )}
   391→    </div>
   392→  );
   393→}

   395→/* ------------------------------------------------------------------ */
   396→/*  Reviews Section                                                    */
   397→/* ------------------------------------------------------------------ */
   398→function ReviewSection({ productId }: { productId: string }) {
   399→  const { data, isLoading } = useQuery({
   400→    queryKey: ['reviews', productId],
   401→    queryFn: async () => {
   402→      const res = await fetch(`/api/products/${productId}/reviews?limit=50`);
   403→      if (!res.ok) throw new Error('Failed to load reviews');
   404→      return res.json() as Promise<{ reviews: { id: string; userName: string; rating: number; title?: string | null; comment: string; isVerified: boolean; createdAt: string }[]; total: number; averageRating: number }>;
   405→    },
   406→  });
   407→
   408→  const reviews = data?.reviews ?? [];
   409→  const total = data?.total ?? 0;
   410→  const averageRating = data?.averageRating ?? 0;
   411→
   412→  return (
   413→    <div>
   414→      <ReviewForm productId={productId} />
   415→
   416→      {/* Summary */}
   417→      <div className="flex items-center gap-3 mb-5">
   418→        <div className="flex items-center gap-1.5">
   419→          <Star className="h-5 w-5 fill-amber-400 text-amber-400" />
   420→          <span className="text-lg font-bold text-[#0F172A]">
   421→            {averageRating > 0 ? averageRating.toFixed(1) : '0.0'}
   422→          </span>
   423→        </div>
   424→        <span className="text-sm text-gray-500">
   425→          Based on {total} review{total !== 1 ? 's' : ''}
   426→        </span>
   427→      </div>
   428→
   429→      {/* Review list */}
   430→      {isLoading ? (
   431→        <div className="space-y-4">
   432→          {[1, 2, 3].map((i) => (
   433→            <Card key={i} className="border-gray-100">
   434→              <CardContent className="p-5">
   435→                <div className="flex items-center gap-3">
   436→                  <Skeleton className="h-10 w-10 rounded-full" />
   437→                  <div className="space-y-2">
   438→                    <Skeleton className="h-4 w-32" />
   439→                    <Skeleton className="h-3 w-24" />
   440→                  </div>
   441→                </div>
   442→                <Skeleton className="h-4 w-full mt-3" />
   443→                <Skeleton className="h-4 w-3/4 mt-2" />
   444→              </CardContent>
   445→            </Card>
   446→          ))}
   447→        </div>
   448→      ) : reviews.length === 0 ? (
   449→        <p className="text-sm text-gray-500 text-center py-8">No reviews yet. Be the first to share your experience!</p>
   450→      ) : (
   451→        <div className="space-y-4 max-h-[500px] overflow-y-auto pr-2">
   452→          {reviews.map((review) => (
   453→            <Card key={review.id} className="border-gray-100">
   454→              <CardContent className="p-5">
   455→                <div className="flex items-start justify-between">
   456→                  <div className="flex items-center gap-3">
   457→                    <Avatar className="h-10 w-10 bg-[#0F172A]">
   458→                      <AvatarFallback className="text-white text-sm font-semibold">
   459→                        {review.userName
   460→                          .split(' ')
   461→                          .map((n) => n[0])
   462→                          .join('')
   463→                          .slice(0, 2)
   464→                          .toUpperCase()}
   465→                      </AvatarFallback>
   466→                    </Avatar>
   467→                    <div>
   468→                      <div className="flex items-center gap-2">
   469→                        <p className="font-semibold text-sm text-[#0F172A]">
   470→                          {review.userName}
   471→                        </p>
   472→                        {review.isVerified && (
   473→                          <Badge className="bg-green-100 text-green-700 hover:bg-green-100 text-[10px] px-1.5 py-0">
   474→                            Verified
   475→                          </Badge>
   476→                        )}
   477→                      </div>
   478→                      <p className="text-xs text-gray-400">
   479→                        {formatDistanceToNow(new Date(review.createdAt), { addSuffix: true })}
   480→                      </p>
   481→                    </div>
   482→                  </div>
   483→                  <StarRating rating={review.rating} size={14} />
   484→                </div>
   485→                {review.title && (
   486→                  <p className="mt-2 text-sm font-semibold text-[#0F172A]">
   487→                    {review.title}
   488→                  </p>
   489→                )}
   490→                <p className="mt-2 text-sm text-gray-600 leading-relaxed">
   491→                  {review.comment}
   492→                </p>
   493→              </CardContent>
   494→            </Card>
   495→          ))}
   496→        </div>
   497→      )}
   498→    </div>
   499→  );
   500→}
