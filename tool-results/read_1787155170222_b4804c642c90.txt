   879→function ProductDialog({
   880→  open,
   881→  onOpenChange,
   882→}: {
   883→  open: boolean;
   884→  onOpenChange: (open: boolean) => void;
   885→}) {
   886→  const queryClient = useQueryClient();
   887→  const [form, setForm] = useState({
   888→    name: '', description: '', price: '', originalPrice: '', image: '', images: '',
   889→    categoryId: '', sellerId: '', stock: '100', featured: false, status: 'ACTIVE',
   890→  });
   891→  const [submitting, setSubmitting] = useState(false);
   892→
   893→  const { data: categories } = useQuery<Category[]>({
   894→    queryKey: ['categories'],
   895→    queryFn: () => fetch('/api/categories').then((r) => r.json()),
   896→  });
   897→
   898→  const { data: sellers } = useQuery<AdminSeller[]>({
   899→    queryKey: ['admin-sellers-dialog'],
   900→    queryFn: () => fetch('/api/admin/sellers').then((r) => r.json()),
   901→  });
   902→
   903→  const updateField = useCallback((key: string, value: string | boolean) => {
   904→    setForm((prev) => ({ ...prev, [key]: value }));
   905→  }, []);
   906→
   907→  const handleSubmit = async () => {
   908→    if (!form.name || !form.description || !form.price || !form.image || !form.categoryId || !form.sellerId) {
   909→      toast.error('Please fill all required fields');
   910→      return;
   911→    }
   912→
   913→    setSubmitting(true);
   914→    try {
   915→      const res = await fetch('/api/admin/products', {
   916→        method: 'POST',
   917→        headers: { 'Content-Type': 'application/json' },
   918→        body: JSON.stringify(form),
   919→      });
   920→      if (res.ok) {
   921→        toast.success('Product created successfully');
   922→        queryClient.invalidateQueries({ queryKey: ['admin-products'] });
   923→        onOpenChange(false);
   924→        setForm({
   925→          name: '', description: '', price: '', originalPrice: '', image: '', images: '',
   926→          categoryId: '', sellerId: '', stock: '100', featured: false, status: 'ACTIVE',
   927→        });
   928→      } else {
   929→        const err = await res.json();
   930→        toast.error(err.error || 'Failed to create product');
   931→      }
   932→    } catch {
   933→      toast.error('Failed to create product');
   934→    } finally {
   935→      setSubmitting(false);
   936→    }
   937→  };

   940→    <Dialog open={open} onOpenChange={onOpenChange}>

   950→              <Label>Product Name *</Label>
   951→              <Input
   952→                value={form.name}
   953→                onChange={(e) => updateField('name', e.target.value)}
   954→                placeholder="Product name"
   955→              />
   956→            </div>
   957→            <div className="space-y-2">
   958→              <Label>Price (KES) *</Label>

   960→                type="number"
   961→                value={form.price}
   962→                onChange={(e) => updateField('price', e.target.value)}
   963→                placeholder="0"
   964→              />
   965→            </div>
