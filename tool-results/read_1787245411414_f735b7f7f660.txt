   160→function StepIndicator({ currentStep }: { currentStep: Step }) {
   161→  return (
   162→    <div className="w-full max-w-2xl mx-auto">
   163→      <div className="flex items-center justify-between">
   164→        {STEP_LABELS.map((label, idx) => {
   165→          const step = (idx + 1) as Step;
   166→          const isCompleted = step < currentStep;
   167→          const isActive = step === currentStep;
   168→          const isUpcoming = step > currentStep;
   169→
   170→          return (
   171→            <div key={label} className="flex items-center flex-1 last:flex-none">
   172→              <div className="flex flex-col items-center gap-1.5">
   173→                <div
   174→                  className={`w-9 h-9 rounded-full flex items-center justify-center text-sm font-semibold transition-all duration-300 ${
   175→                    isCompleted
   176→                      ? 'bg-[#0F172A] text-white'
   177→                      : isActive
   178→                        ? 'bg-[#F59E0B] text-[#0F172A] ring-4 ring-amber-100'
   179→                        : 'bg-gray-100 text-gray-400'
   180→                  }`}
   181→                >
   182→                  {isCompleted ? <Check className="h-4 w-4" /> : step}
   183→                </div>
   184→                <span
   185→                  className={`text-xs font-medium hidden sm:block ${
   186→                    isActive ? 'text-[#0F172A]' : isCompleted ? 'text-[#0F172A]' : 'text-gray-400'
   187→                  }`}
   188→                >
   189→                  {label}
   190→                </span>
   191→              </div>
   192→              {idx < STEP_LABELS.length - 1 && (
   193→                <div
   194→                  className={`flex-1 h-0.5 mx-3 mt-[-1.25rem] sm:mt-[-1.25rem] transition-colors duration-300 ${
   195→                    step < currentStep ? 'bg-[#0F172A]' : 'bg-gray-200'
   196→                  }`}
   197→                />
   198→              )}
   199→            </div>
   200→          );
   201→        })}
   202→      </div>
   203→    </div>
   204→  );
   205→}

   207→function OrderSummarySidebar({
   208→  items,
   209→  deliveryFee,
   210→  couponDiscount,
   211→  couponCode,
   212→  isFreeShipping,
   213→  onCouponApply,
   214→  onCouponRemove,
   215→}: {
   216→  items: CartItem[];
   217→  deliveryFee: number;
   218→  couponDiscount: number;
   219→  couponCode: string | null;
   220→  isFreeShipping: boolean;
   221→  onCouponApply: (discount: number, code: string, isFreeShipping: boolean) => void;
   222→  onCouponRemove: () => void;
   223→}) {
   224→  const subtotal = items.reduce((sum, i) => sum + i.product.price * i.quantity, 0);
   225→  const platformFee = Math.round(subtotal * 0.02);
   226→  const effectiveDeliveryFee = isFreeShipping ? 0 : deliveryFee;
   227→  const total = Math.max(0, subtotal + effectiveDeliveryFee + platformFee - couponDiscount);
   228→
   229→  return (
   230→    <div className="bg-gray-50 rounded-xl p-6 border border-gray-100 sticky top-24">
   231→      <h3 className="font-semibold text-[#0F172A] text-sm mb-4">Order Summary</h3>
   232→      <div className="space-y-3 max-h-64 overflow-y-auto pr-1">
   233→        {items.map((item) => (
   234→          <div key={item.product.id} className="flex items-center gap-3">
   235→            <div className="relative w-12 h-12 rounded-lg overflow-hidden bg-gray-100 shrink-0">
   236→              <Image
   237→                src={item.product.image}
   238→                alt={item.product.name}
   239→                fill
   240→                className="object-cover"
   241→                sizes="48px"
   242→              />
   243→            </div>
   244→            <div className="flex-1 min-w-0">
   245→              <p className="text-sm font-medium text-[#0F172A] truncate">{item.product.name}</p>
   246→              <p className="text-xs text-gray-500">Qty: {item.quantity}</p>
   247→            </div>
   248→            <p className="text-sm font-semibold text-[#0F172A]">
   249→              KES {(item.product.price * item.quantity).toLocaleString()}
   250→            </p>
   251→          </div>
   252→        ))}
   253→      </div>
   254→      <Separator className="my-4" />
   255→      <div className="space-y-2 text-sm">
   256→        <div className="flex justify-between text-gray-600">
   257→          <span>Subtotal</span>
   258→          <span>KES {subtotal.toLocaleString()}</span>
   259→        </div>
   260→        <div className="flex justify-between text-gray-600">
   261→          <span>Delivery</span>
   262→          <span className={isFreeShipping ? 'line-through text-gray-400' : ''}>
   263→            KES {deliveryFee.toLocaleString()}
   264→          </span>
   265→        </div>
   266→        {isFreeShipping && (
   267→          <div className="flex justify-between text-[#F59E0B] font-medium">
   268→            <span>Free Shipping</span>
   269→            <span>−KES {deliveryFee.toLocaleString()}</span>
   270→          </div>
   271→        )}
   272→        <div className="flex justify-between text-gray-600">
   273→          <span>Platform Fee (2%)</span>
   274→          <span>KES {platformFee.toLocaleString()}</span>
   275→        </div>
   276→        {couponDiscount > 0 && (
   277→          <div className="flex justify-between text-green-600 font-medium">
   278→            <span>Coupon Discount ({couponCode})</span>
   279→            <span>−KES {couponDiscount.toLocaleString()}</span>
   280→          </div>
   281→        )}
   282→        <Separator className="!my-3" />
   283→        <div className="flex justify-between font-bold text-[#0F172A]">
   284→          <span>Total</span>
   285→          <span className="text-base">KES {total.toLocaleString()}</span>
   286→        </div>
   287→      </div>
   288→      <div className="mt-4">
   289→        <CouponInput
   290→          onApply={onCouponApply}
   291→          onRemove={onCouponRemove}
   292→          orderTotal={subtotal}
   293→          appliedCode={couponCode}
   294→        />
   295→      </div>
   296→    </div>
   297→  );

  1088→// Main Page
  1089→// ============================================================
  1090→
  1091→export default function CheckoutPage() {
  1092→  const items = useCartStore((s) => s.items);
  1093→  const clearCart = useCartStore((s) => s.clearCart);
  1094→
  1095→  const [step, setStep] = useState<Step>(1);
  1096→  const [direction, setDirection] = useState(1);
  1097→
  1098→  // Shipping
  1099→  const [shipping, setShipping] = useState<ShippingForm>(initialShipping);
  1100→  const [shippingErrors, setShippingErrors] = useState<ShippingErrors>({});
  1101→
  1102→  // Payment
  1103→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethod>('mpesa');
  1104→  const [mpesaPhone, setMpesaPhone] = useState('');
  1105→  const [cardDetails, setCardDetails] = useState({ number: '', expiry: '', cvv: '' });
  1106→
  1107→  // Coupon
  1108→  const [couponCode, setCouponCode] = useState<string | null>(null);
  1109→  const [couponDiscount, setCouponDiscount] = useState(0);
  1110→  const [isFreeShipping, setIsFreeShipping] = useState(false);
  1111→
  1112→  const handleCouponApply = (discount: number, code: string, freeShipping: boolean) => {
  1113→    setCouponDiscount(discount);
  1114→    setCouponCode(code);
  1115→    setIsFreeShipping(freeShipping);
  1116→  };
  1117→
  1118→  const handleCouponRemove = () => {
  1119→    setCouponDiscount(0);
  1120→    setCouponCode(null);
  1121→    setIsFreeShipping(false);
  1122→  };
  1123→
  1124→  // Order
  1125→  const [orderNumber, setOrderNumber] = useState('');
  1126→  const [isPlacing, setIsPlacing] = useState(false);

  1128→  const deliveryFee = useMemo(() => {

  1130→    return 500;

  1136→
  1137→  const handleShippingChange = (field: keyof ShippingForm, value: string) => {

  1139→    if (shippingErrors[field as keyof ShippingErrors]) {
  1140→      setShippingErrors((prev) => ({ ...prev, [field]: undefined }));

  1142→  };
  1143→
  1144→  const validateShipping = (): boolean => {

  1146→    if (!shipping.fullName.trim()) errs.fullName = 'Full name is required';
  1147→    if (!shipping.email.trim() || !/\S+@\S+\.\S+/.test(shipping.email))
  1148→      errs.email = 'Valid email is required';
  1149→    if (!shipping.phone.trim()) errs.phone = 'Phone number is required';
  1150→    if (!shipping.address1.trim()) errs.address1 = 'Address is required';
  1151→    if (!shipping.city.trim()) errs.city = 'City is required';

  1153→    if (!shipping.postalCode.trim()) errs.postalCode = 'Postal code is required';
  1154→    setShippingErrors(errs);

  1161→  };

  1168→  const goBack = () => {

  1187→    const subtotal = items.reduce((sum, i) => sum + i.product.price * i.quantity, 0);
  1188→    const platformFee = Math.round(subtotal * 0.02);

  1194→        method: 'POST',
  1195→        headers: { 'Content-Type': 'application/json' },

  1198→          items: items.map((i) => ({

  1202→            price: i.product.price,
  1203→            image: i.product.image,

  1205→          })),
  1206→          shippingAddress: JSON.stringify(shipping),
  1207→          paymentMethod,
  1208→          totalAmount: total,
  1209→          deliveryFee: effectiveDeliveryFee,
  1210→          platformFee,
  1211→          couponCode: couponCode || undefined,
  1212→          couponDiscount: couponDiscount || undefined,
  1213→          buyerEmail: shipping.email,
  1214→          buyerName: shipping.fullName,
  1215→          buyerPhone: shipping.phone,
  1216→        }),
  1217→      });
  1218→    } catch {
  1219→      // Even if API fails, still show confirmation for demo purposes
  1220→    }

  1222→    clearCart();

  1224→    setStep(4);

  1226→  };

  1235→        <SimpleNavbar />

  1237→          <EmptyCart />
  1238→        </main>
  1239→        <Footer />
  1240→      </div>
  1241→    );

  1243→
  1244→  return (
  1245→    <div className="min-h-screen flex flex-col bg-white">
  1246→      <SimpleNavbar />
  1247→
  1248→      <main className="flex-1 px-6 md:px-16 lg:px-32 py-8 md:py-12">
  1249→        {/* Step Indicator */}
  1250→        {step < 4 && (
  1251→          <div className="mb-10">
  1252→            <StepIndicator currentStep={step} />

  1256→        {/* Step Content */}

  1258→          <div className="max-w-6xl mx-auto grid grid-cols-1 lg:grid-cols-3 gap-8">
  1259→            <div className="lg:col-span-2">
  1260→              <AnimatePresence mode="wait" custom={direction}>
  1261→                <motion.div

  1265→                  initial="enter"
