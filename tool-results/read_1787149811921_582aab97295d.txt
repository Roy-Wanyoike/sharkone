   921→// ============================================================
   922→// Main Page
   923→// ============================================================
   924→
   925→export default function CheckoutPage() {
   926→  const items = useCartStore((s) => s.items);
   927→  const clearCart = useCartStore((s) => s.clearCart);
   928→
   929→  const [step, setStep] = useState<Step>(1);
   930→  const [direction, setDirection] = useState(1);
   931→
   932→  // Shipping
   933→  const [shipping, setShipping] = useState<ShippingForm>(initialShipping);
   934→  const [shippingErrors, setShippingErrors] = useState<ShippingErrors>({});
   935→
   936→  // Payment
   937→  const [paymentMethod, setPaymentMethod] = useState<PaymentMethod>('mpesa');
   938→  const [mpesaPhone, setMpesaPhone] = useState('');
   939→  const [cardDetails, setCardDetails] = useState({ number: '', expiry: '', cvv: '' });
   940→
   941→  // Order
   942→  const [orderNumber, setOrderNumber] = useState('');
   943→  const [isPlacing, setIsPlacing] = useState(false);
   944→
   945→  const deliveryFee = useMemo(() => {
   946→    if (shipping.city.toLowerCase() === 'nairobi') return 250;
   947→    return 500;
   948→  }, [shipping.city]);
   949→
   950→  // ============================================================
   951→  // Helpers
   952→  // ============================================================
   953→
   954→  const handleShippingChange = (field: keyof ShippingForm, value: string) => {
   955→    setShipping((prev) => ({ ...prev, [field]: value }));
   956→    if (shippingErrors[field as keyof ShippingErrors]) {
   957→      setShippingErrors((prev) => ({ ...prev, [field]: undefined }));
   958→    }
   959→  };
   960→
   961→  const validateShipping = (): boolean => {
   962→    const errs: ShippingErrors = {};
   963→    if (!shipping.fullName.trim()) errs.fullName = 'Full name is required';
   964→    if (!shipping.email.trim() || !/\S+@\S+\.\S+/.test(shipping.email))
   965→      errs.email = 'Valid email is required';
   966→    if (!shipping.phone.trim()) errs.phone = 'Phone number is required';
   967→    if (!shipping.address1.trim()) errs.address1 = 'Address is required';
   968→    if (!shipping.city.trim()) errs.city = 'City is required';
   969→    if (!shipping.state) errs.state = 'County/State is required';
   970→    if (!shipping.postalCode.trim()) errs.postalCode = 'Postal code is required';
   971→    setShippingErrors(errs);
   972→    return Object.keys(errs).length === 0;
   973→  };
   974→
   975→  const generateOrderNumber = () => {
   976→    const digits = Math.floor(100000 + Math.random() * 900000).toString();
   977→    return `SHK-${digits}`;
   978→  };
   979→
   980→  const goNext = () => {
   981→    setDirection(1);
   982→    setStep((prev) => Math.min(prev + 1, 4) as Step);
   983→  };
   984→
   985→  const goBack = () => {
   986→    setDirection(-1);
   987→    setStep((prev) => Math.max(prev - 1, 1) as Step);
   988→  };
   989→
   990→  const handleNext = () => {
   991→    if (step === 1) {
   992→      if (!validateShipping()) return;
   993→      goNext();
   994→    } else if (step === 2) {
   995→      goNext();
   996→    }
   997→  };
   998→
   999→  const handlePlaceOrder = async () => {
  1000→    setIsPlacing(true);
  1001→    const num = generateOrderNumber();
  1002→    setOrderNumber(num);
  1003→
  1004→    const subtotal = items.reduce((sum, i) => sum + i.product.price * i.quantity, 0);
  1005→    const platformFee = Math.round(subtotal * 0.02);
  1006→    const total = subtotal + deliveryFee + platformFee;
  1007→
  1008→    try {
  1009→      await fetch('/api/orders', {
  1010→        method: 'POST',
  1011→        headers: { 'Content-Type': 'application/json' },
  1012→        body: JSON.stringify({
  1013→          orderNumber: num,
  1014→          items: items.map((i) => ({
  1015→            productId: i.product.id,
  1016→            name: i.product.name,
  1017→            quantity: i.quantity,
  1018→            price: i.product.price,
  1019→            image: i.product.image,
  1020→            sellerId: i.product.sellerId || '',
  1021→          })),
  1022→          shippingAddress: JSON.stringify(shipping),
  1023→          paymentMethod,
  1024→          totalAmount: total,
  1025→          deliveryFee,
  1026→          platformFee,
  1027→          buyerEmail: shipping.email,
  1028→          buyerName: shipping.fullName,
  1029→          buyerPhone: shipping.phone,
  1030→        }),
  1031→      });
  1032→    } catch {
  1033→      // Even if API fails, still show confirmation for demo purposes
  1034→    }
  1035→
  1036→    clearCart();
  1037→    setDirection(1);
  1038→    setStep(4);
  1039→    setIsPlacing(false);
  1040→  };
  1041→
  1042→  // ============================================================
  1043→  // Render
  1044→  // ============================================================
  1045→
  1046→  if (items.length === 0 && step !== 4) {
  1047→    return (
  1048→      <div className="min-h-screen flex flex-col bg-white">
  1049→        <SimpleNavbar />
  1050→        <main className="flex-1 px-6 md:px-16 lg:px-32 py-10">
  1051→          <EmptyCart />
  1052→        </main>
  1053→        <Footer />
  1054→      </div>
  1055→    );
  1056→  }
  1057→
  1058→  return (
  1059→    <div className="min-h-screen flex flex-col bg-white">
  1060→      <SimpleNavbar />
  1061→
  1062→      <main className="flex-1 px-6 md:px-16 lg:px-32 py-8 md:py-12">
  1063→        {/* Step Indicator */}
  1064→        {step < 4 && (
  1065→          <div className="mb-10">
  1066→            <StepIndicator currentStep={step} />
  1067→          </div>
  1068→        )}
  1069→
  1070→        {/* Step Content */}
  1071→        {step < 4 && step < 3 && (
  1072→          <div className="max-w-6xl mx-auto grid grid-cols-1 lg:grid-cols-3 gap-8">
  1073→            <div className="lg:col-span-2">
  1074→              <AnimatePresence mode="wait" custom={direction}>
  1075→                <motion.div
  1076→                  key={step}
  1077→                  custom={direction}
  1078→                  variants={slideVariants}
  1079→                  initial="enter"
  1080→                  animate="center"
  1081→                  exit="exit"
  1082→                  transition={{ duration: 0.3, ease: 'easeInOut' }}
  1083→                >
  1084→                  {step === 1 && (
  1085→                    <ShippingStep
  1086→                      form={shipping}
  1087→                      errors={shippingErrors}
  1088→                      onChange={handleShippingChange}
  1089→                    />
  1090→                  )}
  1091→                  {step === 2 && (
  1092→                    <PaymentStep
  1093→                      paymentMethod={paymentMethod}
  1094→                      setPaymentMethod={setPaymentMethod}
  1095→                      mpesaPhone={mpesaPhone}
  1096→                      setMpesaPhone={setMpesaPhone}
  1097→                      cardDetails={cardDetails}
  1098→                      setCardDetails={setCardDetails}
  1099→                    />
  1100→                  )}
  1101→                </motion.div>
  1102→              </AnimatePresence>
  1103→
  1104→              {/* Navigation Buttons */}
  1105→              <div className="flex items-center justify-between mt-8 pt-6 border-t border-gray-100">
  1106→                <Button
  1107→                  variant="ghost"
  1108→                  onClick={goBack}
  1109→                  disabled={step === 1}
  1110→                  className="text-gray-500 hover:text-[#0F172A]"
  1111→                >
  1112→                  <ChevronLeft className="h-4 w-4 mr-1" />
  1113→                  Back
  1114→                </Button>
  1115→                <Button
  1116→                  onClick={handleNext}
  1117→                  className="bg-[#0F172A] hover:bg-slate-800 text-white font-semibold"
  1118→                >
  1119→                  {step === 1 ? 'Continue to Payment' : 'Review Order'}
  1120→                  <ChevronRight className="h-4 w-4 ml-1" />
  1121→                </Button>
  1122→              </div>
  1123→            </div>
  1124→
  1125→            {/* Sidebar */}
  1126→            <div>
  1127→              <OrderSummarySidebar items={items} deliveryFee={deliveryFee} />
  1128→            </div>
  1129→          </div>
  1130→        )}
  1131→
  1132→        {step === 3 && (
  1133→          <div className="max-w-6xl mx-auto">
  1134→            <AnimatePresence mode="wait" custom={direction}>
  1135→              <motion.div
  1136→                key={step}
  1137→                custom={direction}
  1138→                variants={slideVariants}
  1139→                initial="enter"
  1140→                animate="center"
  1141→                exit="exit"
  1142→                transition={{ duration: 0.3, ease: 'easeInOut' }}
