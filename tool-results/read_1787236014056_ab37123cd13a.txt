   570→
   571→  return (
   572→    <motion.div
   573→      initial={{ opacity: 0, x: 20 }}
   574→      animate={{ opacity: 1, x: 0 }}
   575→      exit={{ opacity: 0, x: -20 }}
   576→      transition={{ duration: 0.2 }}
   577→      className="space-y-4"
   578→    >
   579→      {/* Header with search, filter, add button */}
   580→      <div className="flex flex-col sm:flex-row gap-3">
   581→        <div className="relative flex-1">
   582→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-slate-400" />
   583→          <Input
