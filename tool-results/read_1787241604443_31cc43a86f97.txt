   112→   112→function formatCurrency(value: number): string {

   116→   116→function formatDate(dateStr: string): string {

   120→   120→function statusBadgeColor(status: string) {
   121→   121→  switch (status) {
   122→   122→    case 'DELIVERED': return 'bg-emerald-100 text-emerald-700';
   123→   123→    case 'SHIPPED': return 'bg-sky-100 text-sky-700';
   124→   124→    case 'PROCESSING': return 'bg-amber-100 text-amber-700';
   125→   125→    case 'PENDING': return 'bg-gray-100 text-gray-700';
   126→   126→    case 'CANCELLED': return 'bg-red-100 text-red-700';
   127→   127→    case 'CONFIRMED': return 'bg-blue-100 text-blue-700';
   128→   128→    case 'OUT_FOR_DELIVERY': return 'bg-violet-100 text-violet-700';
   129→   129→    case 'REFUNDED': return 'bg-orange-100 text-orange-700';
   130→   130→    default: return 'bg-gray-100 text-gray-700';
   131→   131→  }
   132→   132→}
   133→   133→
   134→   134→function formatStatusLabel(status: string): string {
   135→   135→  return status.replace(/_/g, ' ');

   137→   137→
   138→   138→function txnTypeColor(type: string) {
   139→   139→  switch (type) {
   140→   140→    case 'EARNING': return 'text-emerald-600';
   141→   141→    case 'WITHDRAWAL': return 'text-orange-600';
   142→   142→    case 'REFUND': return 'text-red-600';
   143→   143→    case 'PURCHASE': return 'text-gray-600';
   144→   144→    default: return 'text-gray-600';
   145→   145→  }
   146→   146→}
   147→   147→
   148→   148→function formatTxnType(type: string): string {

   215→   215→
   216→   216→export function SellerDashboard({ sellerId: propSellerId }: SellerDashboardProps) {
   217→   217→  const queryClient = useQueryClient();
   218→   218→
   219→   219→  // Auto-detect seller if not provided
   220→   220→  const autoSellersQuery = useQuery<{ id: string }[]>({
   221→   221→    queryKey: ['sellers-auto-detect'],
   222→   222→    queryFn: () => fetch('/api/admin/sellers').then(r => r.json()),
   223→   223→    enabled: !propSellerId,
   224→   224→  });
   225→   225→
   226→   226→  const sellerId = propSellerId || autoSellersQuery.data?.[0]?.id || '';
   227→   227→  const isLoading = !propSellerId ? autoSellersQuery.isLoading : false;
   228→   228→  const [addProductOpen, setAddProductOpen] = useState(false);
   229→   229→  const [newProduct, setNewProduct] = useState({ name: '', price: '', description: '', category: '', stock: '' });
   230→   230→  const [withdrawAmount, setWithdrawAmount] = useState('');
   231→   231→  const [withdrawOpen, setWithdrawOpen] = useState(false);
   232→   232→
   233→   233→  /* ---- Queries ---- */
   234→   234→
   235→   235→  const statsQuery = useQuery<SellerStats>({
   236→   236→    queryKey: ['seller-stats', sellerId],
   237→   237→    queryFn: () => fetch(`/api/seller/${sellerId}/stats`).then(r => r.json()),
   238→   238→    enabled: !!sellerId,
   239→   239→  });
   240→   240→
   241→   241→  const productsQuery = useQuery<SellerProduct[]>({
   242→   242→    queryKey: ['seller-products', sellerId],
   243→   243→    queryFn: () => fetch(`/api/seller/${sellerId}/products`).then(r => r.json()),
   244→   244→    enabled: !!sellerId,
   245→   245→  });
   246→   246→
   247→   247→  const ordersQuery = useQuery<SellerOrder[]>({
   248→   248→    queryKey: ['seller-orders', sellerId],
   249→   249→    queryFn: () => fetch(`/api/seller/${sellerId}/orders`).then(r => r.json()),
   250→   250→    enabled: !!sellerId,
   251→   251→  });
   252→   252→
   253→   253→  const transactionsQuery = useQuery<SellerTransaction[]>({
   254→   254→    queryKey: ['seller-transactions', sellerId],
   255→   255→    queryFn: () => fetch(`/api/seller/${sellerId}/transactions`).then(r => r.json()),
   256→   256→    enabled: !!sellerId,
   257→   257→  });
   258→   258→
   259→   259→  const categoriesQuery = useQuery<Category[]>({
   260→   260→    queryKey: ['categories'],
   261→   261→    queryFn: () => fetch('/api/categories').then(r => r.json()),
   262→   262→  });
   263→   263→
   264→   264→  /* ---- Mutations ---- */
   265→   265→
   266→   266→  const addProductMutation = useMutation({
   267→   267→    mutationFn: (data: { name: string; price: string; description: string; category: string; stock: string }) =>
   268→   268→      fetch(`/api/seller/${sellerId}/products`, {
   269→   269→        method: 'POST',
   270→   270→        headers: { 'Content-Type': 'application/json' },
   271→   271→        body: JSON.stringify(data),
   272→   272→      }).then(r => r.json()),
   273→   273→    onSuccess: () => {
   274→   274→      queryClient.invalidateQueries({ queryKey: ['seller-products', sellerId] });
   275→   275→      queryClient.invalidateQueries({ queryKey: ['seller-stats', sellerId] });
   276→   276→      toast.success(`"${newProduct.name}" has been added as a draft`);
   277→   277→      setAddProductOpen(false);
   278→   278→      setNewProduct({ name: '', price: '', description: '', category: '', stock: '' });
   279→   279→    },
   280→   280→    onError: () => {
   281→   281→      toast.error('Failed to create product');
   282→   282→    },
   283→   283→  });
   284→   284→
   285→   285→  const withdrawMutation = useMutation({
   286→   286→    mutationFn: (amount: number) =>
   287→   287→      fetch(`/api/seller/${sellerId}/withdraw`, {
   288→   288→        method: 'POST',
   289→   289→        headers: { 'Content-Type': 'application/json' },
   290→   290→        body: JSON.stringify({ amount }),
   291→   291→      }).then(r => {
   292→   292→        if (!r.ok) return r.json().then(d => { throw new Error(d.error || 'Withdrawal failed'); });
   293→   293→        return r.json();
   294→   294→      }),
   295→   295→    onSuccess: () => {
   296→   296→      queryClient.invalidateQueries({ queryKey: ['seller-stats', sellerId] });
   297→   297→      queryClient.invalidateQueries({ queryKey: ['seller-transactions', sellerId] });
   298→   298→      toast.success('Withdrawal request submitted!');
   299→   299→      setWithdrawOpen(false);
   300→   300→      setWithdrawAmount('');
   301→   301→    },
   302→   302→    onError: (err: Error) => {
   303→   303→      toast.error(err.message);
   304→   304→    },
   305→   305→  });
   306→   306→
   307→   307→  const handleAddProduct = () => {
   308→   308→    if (!newProduct.name || !newProduct.price) {
   309→   309→      toast.error('Please fill in product name and price');
   310→   310→      return;
   311→   311→    }
   312→   312→    addProductMutation.mutate(newProduct);
   313→   313→  };
   314→   314→
   315→   315→  const handleWithdraw = () => {
   316→   316→    const amount = parseFloat(withdrawAmount);
   317→   317→    if (!amount || amount <= 0) {
   318→   318→      toast.error('Please enter a valid amount');
   319→   319→      return;
   320→   320→    }
   321→   321→    withdrawMutation.mutate(amount);
   322→   322→  };
   323→   323→
   324→   324→  const categories = categoriesQuery.data || [];
   325→   325→  const stats = statsQuery.data;
   326→   326→  const products = productsQuery.data || [];
   327→   327→  const orders = ordersQuery.data || [];
   328→   328→  const transactions = transactionsQuery.data || [];
   329→   329→
   330→   330→  const statsCards = stats
   331→   331→    ? [
   332→   332→        { label: 'Total Revenue', value: formatCurrency(stats.totalRevenue), change: '+12.5%', icon: <DollarSign className="h-5 w-5" />, color: 'bg-amber-50 text-amber-700' },
   333→   333→        { label: 'Total Orders', value: String(stats.totalOrders), change: '+8.2%', icon: <ShoppingBag className="h-5 w-5" />, color: 'bg-emerald-50 text-emerald-700' },
   334→   334→        { label: 'Products Listed', value: String(stats.totalProducts), change: `+${stats.totalProducts > 0 ? '1' : '0'}`, icon: <Package className="h-5 w-5" />, color: 'bg-gray-100 text-gray-700' },
   335→   335→        { label: 'Rating', value: stats.rating > 0 ? stats.rating.toFixed(1) : 'N/A', change: stats.rating > 0 ? `+${(stats.rating * 0.02).toFixed(1)}` : '', icon: <Star className="h-5 w-5" />, color: 'bg-amber-50 text-amber-700' },
   336→   336→      ]
   337→   337→    : [];
   338→   338→
   339→   339→  return (
   340→   340→    <div className="px-4 md:px-16 lg:px-32 py-8">
   341→   341→      {/* Header */}
   342→   342→      <div className="mb-8">
   343→   343→        <h1 className="text-2xl md:text-3xl font-bold text-gray-900">Seller Dashboard</h1>
   344→   344→        <p className="text-gray-500 text-sm mt-1">Manage your store, products, and earnings</p>
   345→   345→      </div>
   346→   346→
   347→   347→      <Tabs defaultValue="overview" className="w-full">
   348→   348→        <TabsList className="mb-6">
   349→   349→          <TabsTrigger value="overview">Overview</TabsTrigger>
   350→   350→          <TabsTrigger value="products">Products</TabsTrigger>
   351→   351→          <TabsTrigger value="orders">Orders</TabsTrigger>
   352→   352→          <TabsTrigger value="wallet">Wallet</TabsTrigger>
   353→   353→        </TabsList>
   354→   354→
   355→   355→        {/* ---- Overview Tab ---- */}
   356→   356→        <TabsContent value="overview">
   357→   357→          {statsQuery.isLoading ? (
   358→   358→            <div className="grid grid-cols-2 lg:grid-cols-4 gap-4 mb-8">
   359→   359→              {Array.from({ length: 4 }).map((_, i) => <StatCardSkeleton key={i} />)}
   360→   360→            </div>
   361→   361→          ) : statsQuery.error ? (
   362→   362→            <div className="bg-red-50 border border-red-200 rounded-xl p-6 mb-8 flex items-center gap-3">
   363→   363→              <AlertCircle className="h-5 w-5 text-red-500" />
   364→   364→              <p className="text-sm text-red-700">Failed to load dashboard stats. Please try again.</p>
   365→   365→            </div>
   366→   366→          ) : (
   367→   367→            <div className="grid grid-cols-2 lg:grid-cols-4 gap-4 mb-8">
   368→   368→              {statsCards.map((stat, i) => (
   369→   369→                <motion.div
   370→   370→                  key={stat.label}
   371→   371→                  initial={{ opacity: 0, y: 20 }}
   372→   372→                  animate={{ opacity: 1, y: 0 }}
   373→   373→                  transition={{ delay: i * 0.05 }}
   374→   374→                  className="bg-white border border-gray-200 rounded-xl p-5"
   375→   375→                >
   376→   376→                  <div className="flex items-center justify-between mb-3">
   377→   377→                    <div className={`p-2 rounded-lg ${stat.color}`}>{stat.icon}</div>
   378→   378→                    {stat.change && (
   379→   379→                      <span className="text-xs font-medium text-emerald-600 flex items-center gap-0.5">
   380→   380→                        <TrendingUp className="h-3 w-3" />
   381→   381→                        {stat.change}
   382→   382→                      </span>
   383→   383→                    )}
   384→   384→                  </div>
   385→   385→                  <p className="text-2xl font-bold text-gray-900">{stat.value}</p>
   386→   386→                  <p className="text-xs text-gray-500 mt-1">{stat.label}</p>
   387→   387→                </motion.div>
   388→   388→              ))}
   389→   389→            </div>
   390→   390→          )}
   391→   391→
   392→   392→          {/* Recent Orders Table */}
   393→   393→          {ordersQuery.isLoading ? (
   394→   394→            <TableSkeleton rows={5} />
   395→   395→          ) : orders.length === 0 ? (
   396→   396→            <div className="bg-white border border-gray-200 rounded-xl p-12 text-center">
   397→   397→              <ShoppingBag className="h-12 w-12 text-gray-300 mx-auto mb-3" />
   398→   398→              <p className="text-gray-500 text-sm">No orders yet. Your orders will appear here.</p>
   399→   399→            </div>
   400→   400→          ) : (
   401→   401→            <div className="bg-white border border-gray-200 rounded-xl overflow-hidden">
   402→   402→              <div className="px-6 py-4 border-b border-gray-100">
   403→   403→                <h2 className="font-semibold text-gray-900">Recent Orders</h2>
   404→   404→              </div>
   405→   405→              <div className="overflow-x-auto">
   406→   406→                <table className="w-full text-sm">
   407→   407→                  <thead>
   408→   408→                    <tr className="border-b border-gray-100 text-left text-xs text-gray-500 uppercase tracking-wider">
   409→   409→                      <th className="px-6 py-3">Order #</th>
   410→   410→                      <th className="px-6 py-3">Customer</th>
   411→   411→                      <th className="px-6 py-3">Amount</th>
   412→   412→                      <th className="px-6 py-3">Status</th>
   413→   413→                      <th className="px-6 py-3">Date</th>
   414→   414→                    </tr>
   415→   415→                  </thead>
   416→   416→                  <tbody className="divide-y divide-gray-50">
   417→   417→                    {orders.slice(0, 5).map((order) => (
   418→   418→                      <tr key={order.id} className="hover:bg-gray-50 transition">
   419→   419→                        <td className="px-6 py-3 font-medium text-gray-900">{order.orderNumber}</td>
   420→   420→                        <td className="px-6 py-3 text-gray-600">{order.buyerName}</td>
   421→   421→                        <td className="px-6 py-3 font-medium text-gray-900">{formatCurrency(order.total)}</td>
   422→   422→                        <td className="px-6 py-3">
   423→   423→                          <Badge variant="secondary" className={`text-[10px] font-semibold ${statusBadgeColor(order.status)}`}>
   424→   424→                            {formatStatusLabel(order.status)}
   425→   425→                          </Badge>
   426→   426→                        </td>
   427→   427→                        <td className="px-6 py-3 text-gray-500">{formatDate(order.createdAt)}</td>
   428→   428→                      </tr>
   429→   429→                    ))}
   430→   430→                  </tbody>
   431→   431→                </table>
   432→   432→              </div>
   433→   433→            </div>
   434→   434→          )}
   435→   435→        </TabsContent>
   436→   436→
   437→   437→        {/* ---- Products Tab ---- */}
   438→   438→        <TabsContent value="products">
   439→   439→          <div className="flex items-center justify-between mb-6">
   440→   440→            <h2 className="font-semibold text-gray-900 text-lg">Your Products</h2>
   441→   441→            <Dialog open={addProductOpen} onOpenChange={setAddProductOpen}>
   442→   442→              <DialogTrigger asChild>
   443→   443→                <Button className="bg-[#F59E0B] hover:bg-amber-600 text-[#0F172A] font-semibold rounded-lg">
   444→   444→                  <Plus className="h-4 w-4 mr-2" />
   445→   445→                  Add Product
   446→   446→                </Button>
   447→   447→              </DialogTrigger>
   448→   448→              <DialogContent className="sm:max-w-lg">
   449→   449→                <DialogHeader>
   450→   450→                  <DialogTitle>Add New Product</DialogTitle>
   451→   451→                </DialogHeader>
