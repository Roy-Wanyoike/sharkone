   257→   257→
   258→   258→function StatusStepper({ current }: { current: DeliveryStep }) {
   259→   259→  const currentIndex = allSteps.indexOf(current);
   260→   260→
   261→   261→  return (
   262→   262→    <div className="flex items-center gap-0 w-full mt-4">
   263→   263→      {allSteps.map((step, i) => {
   264→   264→        const isCompleted = i < currentIndex;
   265→   265→        const isCurrent = i === currentIndex;
   266→   266→        const isLast = i === allSteps.length - 1;
   267→   267→
   268→   268→        return (
   269→   269→          <div key={step} className="flex items-center flex-1 min-w-0">
   270→   270→            <div className="flex flex-col items-center shrink-0">
   271→   271→              <div
   272→   272→                className={`w-7 h-7 rounded-full flex items-center justify-center text-[10px] font-bold transition-colors ${
   273→   273→                  isCompleted
   274→   274→                    ? 'bg-emerald-500 text-white'
   275→   275→                    : isCurrent
   276→   276→                      ? 'bg-[#F59E0B] text-white ring-4 ring-amber-100'
   277→   277→                      : 'bg-gray-200 text-gray-400'
   278→   278→                }`}
   279→   279→              >
   280→   280→                {isCompleted ? <CheckCircle2 className="h-4 w-4" /> : i + 1}
   281→   281→              </div>
   282→   282→              <span className={`text-[9px] mt-1 text-center leading-tight hidden sm:block ${isCurrent ? 'text-amber-700 font-semibold' : isCompleted ? 'text-emerald-600' : 'text-gray-400'}`}>
   283→   283→                {stepLabels[step]}
   284→   284→              </span>
   285→   285→            </div>
   286→   286→            {!isLast && (
   287→   287→              <div className={`flex-1 h-0.5 mx-1 mt-[-12px] sm:mt-0 ${i < currentIndex ? 'bg-emerald-500' : 'bg-gray-200'}`} />
   288→   288→            )}
   289→   289→          </div>
   290→   290→        );
   291→   291→      })}
   292→   292→    </div>
   293→   293→  );
   294→   294→}
